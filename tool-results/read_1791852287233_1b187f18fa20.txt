    74→export function StartTakeDialog({ open, onOpenChange, onStarted }: {
    75→  open: boolean
    76→  onOpenChange: (v: boolean) => void
    77→  onStarted: (takeId: string) => void
    78→}) {
    79→  const [title, setTitle] = useState('')
    80→  const [scope, setScope] = useState<'all' | 'category'>('all')
    81→  const [categoryId, setCategoryId] = useState('')
    82→  const [note, setNote] = useState('')
    83→  const [categories, setCategories] = useState<Array<{ id: string; name: string; _count: { medicines: number } }>>([])
    84→  const [submitting, setSubmitting] = useState(false)
    85→
    86→  useEffect(() => {
    87→    if (!open) return
    88→    setTitle('')
    89→    setScope('all')
    90→    setCategoryId('')
    91→    setNote('')
    92→    fetch('/api/categories', { cache: 'no-store' })
    93→      .then((r) => (r.ok ? r.json() : { categories: [] }))
    94→      .then((j) => setCategories(j.categories ?? []))
    95→      .catch(() => setCategories([]))
    96→  }, [open])
    97→
    98→  const start = async () => {
    99→    setSubmitting(true)
   100→    try {
   101→      const res = await fetch('/api/stock-takes', {
   102→        method: 'POST',
   103→        headers: { 'Content-Type': 'application/json' },
   104→        body: JSON.stringify({ title: title.trim(), scope, categoryId: scope === 'category' ? categoryId : null, note: note.trim() || null }),
   105→      })
   106→      const json = await res.json()
   107→      if (!res.ok) {
   108→        toast({ title: 'Cannot start stock take', description: json.error ?? 'Unknown error', variant: 'destructive' })
   109→        return
   110→      }
   111→      toast({ title: 'Count session started', description: `${json.take.stockTakeNo} — ${json.aggregates.totalItems} products snapshotted at today's system stock.` })
   112→      onOpenChange(false)
   113→      onStarted(json.take.id)
   114→    } finally {
   115→      setSubmitting(false)
   116→    }
   117→  }
   118→
   119→  return (
   120→    <Dialog open={open} onOpenChange={onOpenChange}>
   121→      <DialogContent className="sm:max-w-md">
   122→        <DialogHeader>
   123→          <DialogTitle className="flex items-center gap-2">
   124→            <PlayCircle className="h-5 w-5 text-teal-600" /> Start Stock Take
   125→          </DialogTitle>
   126→          <DialogDescription>
   127→            A new count session snapshots the current system stock of every product in scope. Enter physical counts against it —
   128→            variances are calculated as counted minus system, positive or negative.
   129→          </DialogDescription>
   130→        </DialogHeader>
   131→        <div className="space-y-4">
   132→          <div className="space-y-1.5">
   133→            <Label>Count title</Label>
   134→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Monthly shelf count" aria-label="Count title" />
   135→          </div>
   136→          <div className="space-y-1.5">
   137→            <Label>Scope</Label>
   138→            <Select value={scope} onValueChange={(v) => setScope(v as 'all' | 'category')}>
   139→              <SelectTrigger aria-label="Count scope"><SelectValue /></SelectTrigger>
   140→              <SelectContent>
   141→                <SelectItem value="all">All active products</SelectItem>
   142→                <SelectItem value="category">Single category</SelectItem>
   143→              </SelectContent>
   144→            </Select>
   145→          </div>
   146→          {scope === 'category' && (
   147→            <div className="space-y-1.5">
   148→              <Label>Category to count</Label>
   149→              <Select value={categoryId} onValueChange={setCategoryId}>
   150→                <SelectTrigger aria-label="Category"><SelectValue placeholder="Pick a category" /></SelectTrigger>
   151→                <SelectContent>
   152→                  {categories.map((c) => (
   153→                    <SelectItem key={c.id} value={c.id}>{c.name} ({c._count.medicines})</SelectItem>
   154→                  ))}
   155→                </SelectContent>
   156→              </Select>
   157→            </div>
   158→          )}
   159→          <div className="space-y-1.5">
   160→            <Label>Note (optional)</Label>
   161→            <Input value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. blind count, shelf A–D" />
   162→          </div>
   163→          <Button className="w-full rounded-full bg-teal-600 hover:bg-teal-700" size="lg" onClick={start} disabled={submitting || !title.trim() || (scope === 'category' && !categoryId)}>
   164→            {submitting ? 'Snapshotting stock…' : 'Start Count Session'}
   165→          </Button>
   166→        </div>
   167→      </DialogContent>
   168→    </Dialog>
   169→  )
   170→}

   176→type LineFilter = 'all' | 'pending' | 'counted' | 'variance'
   177→
   178→export function CountSheetDialog({ open, onOpenChange, takeId, onChanged, onComplete }: {
   179→  open: boolean
   180→  onOpenChange: (v: boolean) => void
   181→  takeId: string
   182→  onChanged: () => void
   183→  onComplete: (take: StockTakeRec) => void
   184→}) {
   185→  const [take, setTake] = useState<StockTakeRec | null>(null)
   186→  const [items, setItems] = useState<StockTakeItemRec[]>([])
   187→  const [loading, setLoading] = useState(true)
   188→  const [search, setSearch] = useState('')
   189→  const [filter, setFilter] = useState<LineFilter>('all')
   190→  const [drafts, setDrafts] = useState<Record<string, string>>({})
   191→  const [savingIds, setSavingIds] = useState<Set<string>>(new Set())
   192→  const [savedIds, setSavedIds] = useState<Set<string>>(new Set())
   193→
   194→  const load = useCallback(async () => {
   195→    try {
   196→      const res = await fetch(`/api/stock-takes/${takeId}`, { cache: 'no-store' })
   197→      if (!res.ok) return
   198→      const json = await res.json()
   199→      setTake(json.take)
   200→      setItems(json.take.items ?? [])
   201→      // Seed drafts with existing counts
   202→      const d: Record<string, string> = {}
   203→      for (const it of json.take.items ?? []) {
   204→        if (it.countedQty !== null) d[it.id] = String(it.countedQty)
   205→      }
   206→      setDrafts(d)
   207→    } finally {
   208→      setLoading(false)
   209→    }
   210→  }, [takeId])
   211→
   212→  useEffect(() => {
   213→    if (!open) return
   214→    setLoading(true)
   215→    setSearch('')
   216→    setFilter('all')
   217→    setSavedIds(new Set())
   218→    load()
   219→  }, [open, load])
   220→
   221→  const saveLine = useCallback(async (item: StockTakeItemRec) => {
   222→    const raw = drafts[item.id]
   223→    if (raw === undefined) return
   224→    const qty = Number(raw)
   225→    if (raw === '' || Number.isNaN(qty) || qty < 0 || !Number.isInteger(qty)) {
   226→      toast({ title: 'Invalid count', description: 'Enter a whole number of 0 or more.', variant: 'destructive' })
   227→      return
   228→    }
   229→    if (item.countedQty === qty) return
   230→    setSavingIds((s) => new Set(s).add(item.id))
   231→    try {
   232→      const res = await fetch(`/api/stock-takes/${takeId}/count`, {
   233→        method: 'PUT',
   234→        headers: { 'Content-Type': 'application/json' },
   235→        body: JSON.stringify({ itemId: item.id, countedQty: qty }),
   236→      })
   237→      const json = await res.json()
   238→      if (!res.ok) {
   239→        toast({ title: 'Could not save count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   240→        return
   241→      }
   242→      setItems((list) => list.map((i) => (i.id === item.id ? { ...i, ...json.item } : i)))
   243→      setTake((t) => (t ? { ...t, ...json.aggregates } : t))
   244→      setSavedIds((s) => new Set(s).add(item.id))
   245→      setSavingIds((s) => { const n = new Set(s); n.delete(item.id); return n })
   246→      onChanged()
   247→    } catch {
   248→      setSavingIds((s) => { const n = new Set(s); n.delete(item.id); return n })
   249→      toast({ title: 'Network error', description: 'Count not saved — try again.', variant: 'destructive' })
   250→    }
   251→  }, [drafts, takeId, onChanged])
   252→
   253→  const filtered = useMemo(() => {
   254→    let list = items
   255→    const q = search.trim().toLowerCase()
   256→    if (q) {
   257→      list = list.filter((i) =>
   258→        i.medicineName.toLowerCase().includes(q) ||
   259→        i.sku.toLowerCase().includes(q) ||
   260→        (i.category ?? '').toLowerCase().includes(q))
   261→    }
   262→    if (filter === 'pending') list = list.filter((i) => i.countedQty === null)
   263→    if (filter === 'counted') list = list.filter((i) => i.countedQty !== null)
   264→    if (filter === 'variance') list = list.filter((i) => (i.varianceQty ?? 0) !== 0)
   265→    return list
   266→  }, [items, search, filter])
   267→
   268→  const progressPct = take && take.totalItems > 0 ? (take.countedItems / take.totalItems) * 100 : 0
   269→  const allCounted = take ? take.countedItems === take.totalItems && take.totalItems > 0 : false
   270→
   271→  return (
   272→    <Dialog open={open} onOpenChange={onOpenChange}>
   273→      <DialogContent className="max-w-5xl">
   274→        <DialogHeader>
   275→          <DialogTitle className="flex flex-wrap items-center gap-2">
   276→            <PackageSearch className="h-5 w-5 text-teal-600" />
   277→            Physical Count — {take?.stockTakeNo ?? '…'}
   278→            {take && <span className="text-sm font-normal text-muted-foreground">{take.title}</span>}
   279→          </DialogTitle>
   280→          <DialogDescription>
   281→            Type the quantity you physically count on the shelf. Counts save automatically on Enter or when you leave the field.
   282→          </DialogDescription>
   283→        </DialogHeader>
   284→
   285→        {loading ? (
   286→          <div className="space-y-2 py-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}</div>
   287→        ) : (
   288→          <>
   289→            {/* Progress + live variance header */}
   290→            <div className="rounded-xl border bg-teal-600/5 p-4">
   291→              <div className="flex flex-wrap items-center justify-between gap-3">
   292→                <div className="min-w-48 flex-1">
   293→                  <div className="flex items-center justify-between text-xs">
   294→                    <span className="font-semibold">{take?.countedItems ?? 0} / {take?.totalItems ?? 0} lines counted</span>
   295→                    <span className="text-muted-foreground">{Math.round(progressPct)}%</span>
   296→                  </div>
   297→                  <Progress value={progressPct} className="mt-2 h-2" aria-label="Count progress" />
   298→                </div>
   299→                <div className="flex flex-wrap items-center gap-2 text-xs">
   300→                  <span className="rounded-full bg-emerald-50 px-2.5 py-1 font-semibold text-emerald-700 dark:bg-emerald-500/10 dark:text-emerald-400">
   301→                    Surplus {take?.positiveLines ?? 0}
   302→                  </span>
   303→                  <span className="rounded-full bg-red-50 px-2.5 py-1 font-semibold text-red-700 dark:bg-red-500/10 dark:text-red-400">
   304→                    Short {take?.negativeLines ?? 0}
   305→                  </span>
   306→                  <span className="rounded-full bg-muted px-2.5 py-1 font-semibold">
   307→                    Net {take ? `${take.varianceQty > 0 ? '+' : ''}${take.varianceQty}` : '0'} units · {take ? `${take.varianceValue >= 0 ? '+' : '−'}${fmtMoney(Math.abs(take.varianceValue)).replace(/^[^0-9]*/, '')}` : ''} value
   308→                  </span>
   309→                </div>
   310→              </div>
   311→            </div>
   312→
   313→            {/* Filters */}
   314→            <div className="flex flex-wrap items-center gap-2">
   315→              <div className="relative flex-1 min-w-48">
   316→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   317→                <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search product, SKU, category…" className="pl-9" aria-label="Search count lines" />
   318→              </div>
   319→              <div className="flex gap-1">
   320→                {([
   321→                  { k: 'all', label: 'All' },
   322→                  { k: 'pending', label: 'Not counted' },
   323→                  { k: 'counted', label: 'Counted' },
   324→                  { k: 'variance', label: 'Variance' },
   325→                ] as Array<{ k: LineFilter; label: string }>).map((f) => (
   326→                  <Button key={f.k} size="sm" variant="outline"
   327→                    className={`rounded-full text-xs ${filter === f.k ? 'border-teal-500 bg-teal-50 text-teal-700 dark:bg-teal-500/10 dark:text-teal-300' : ''}`}
   328→                    onClick={() => setFilter(f.k)}>
   329→                    {f.label}
   330→                  </Button>
   331→                ))}
   332→              </div>
   333→            </div>
   334→
   335→            {/* Lines */}
   336→            <div className="max-h-[46vh] overflow-y-auto rounded-xl border">
   337→              <Table>
   338→                <TableHeader className="sticky top-0 z-10 bg-card">
   339→                  <TableRow className="hover:bg-transparent">
   340→                    <TableHead>Product</TableHead>
   341→                    <TableHead className="text-center">System</TableHead>
   342→                    <TableHead className="w-28 text-center">Counted</TableHead>
   343→                    <TableHead className="text-center">Variance</TableHead>
   344→                    <TableHead className="hidden text-right md:table-cell">Value Impact</TableHead>
   345→                    <TableHead className="w-10"></TableHead>
   346→                  </TableRow>
   347→                </TableHeader>
   348→                <TableBody>
   349→                  {filtered.length === 0 && (
   350→                    <TableRow><TableCell colSpan={6} className="py-10 text-center text-muted-foreground">No lines match this filter.</TableCell></TableRow>
   351→                  )}
   352→                  {filtered.map((item) => {
   353→                    const draft = drafts[item.id] ?? ''
   354→                    const preview = draft === '' ? null : Number(draft)
   355→                    const liveVariance = preview !== null && !Number.isNaN(preview) ? preview - item.systemQty : null
   356→                    return (
   357→                      <TableRow key={item.id} className="hover:bg-muted/30">
   358→                        <TableCell>
   359→                          <p className="max-w-56 truncate font-medium">{item.medicineName}</p>
   360→                          <p className="text-xs text-muted-foreground">
   361→                            {item.sku} · {item.unit}{item.location ? ` · ${item.location}` : ''}
   362→                          </p>
   363→                        </TableCell>
   364→                        <TableCell className="text-center font-semibold tabular-nums">{item.systemQty}</TableCell>
   365→                        <TableCell>
   366→                          <Input
   367→                            type="number"
   368→                            min={0}
   369→                            step={1}
   370→                            inputMode="numeric"
   371→                            value={draft}
   372→                            onChange={(e) => {
   373→                              setDrafts((d) => ({ ...d, [item.id]: e.target.value }))
   374→                              setSavedIds((s) => { const n = new Set(s); n.delete(item.id); return n })
   375→                            }}
   376→                            onBlur={() => saveLine(item)}
   377→                            onKeyDown={(e) => { if (e.key === 'Enter') (e.target as HTMLInputElement).blur() }}
   378→                            placeholder="—"
   379→                            className="h-9 text-center font-semibold tabular-nums"
   380→                            aria-label={`Counted quantity for ${item.medicineName}`}
   381→                          />
   382→                        </TableCell>
   383→                        <TableCell className="text-center">
   384→                          {liveVariance !== null
   385→                            ? <VarianceBadge variance={liveVariance} />
   386→                            : item.countedQty !== null
   387→                              ? <VarianceBadge variance={item.varianceQty ?? 0} />
   388→                              : <span className="text-xs text-muted-foreground">pending</span>}
   389→                        </TableCell>
   390→                        <TableCell className="hidden text-right md:table-cell">
   391→                          {item.varianceValue != null && (item.varianceQty ?? 0) !== 0
   392→                            ? <VarianceValueBadge value={item.varianceValue} />
   393→                            : <span className="text-xs text-muted-foreground">—</span>}
   394→                        </TableCell>
   395→                        <TableCell>
   396→                          {savingIds.has(item.id) ? (
   397→                            <Loader2 className="h-4 w-4 animate-spin text-teal-600" aria-label="Saving" />
   398→                          ) : savedIds.has(item.id) ? (
   399→                            <Check className="h-4 w-4 text-emerald-500" aria-label="Saved" />
   400→                          ) : null}
   401→                        </TableCell>
   402→                      </TableRow>
   403→                    )
   404→                  })}
   405→                </TableBody>
   406→              </Table>
   407→            </div>
   408→
   409→            {/* Footer actions */}
   410→            <div className="flex flex-wrap items-center justify-between gap-3">
   411→              <p className="text-xs text-muted-foreground">
   412→                {allCounted
   413→                  ? 'All lines counted — ready to complete.'
   414→                  : `${(take?.totalItems ?? 0) - (take?.countedItems ?? 0)} line(s) still to count.`}
   415→              </p>
   416→              <div className="flex gap-2">
   417→                <Button variant="outline" className="rounded-full" onClick={() => onOpenChange(false)}>Close</Button>
   418→                <Button
   419→                  className="rounded-full bg-teal-600 hover:bg-teal-700"
   420→                  disabled={!allCounted || !take}
   421→                  onClick={() => { if (take) { onOpenChange(false); onComplete(take) } }}
   422→                >
   423→                  <CheckCircle2 className="h-4 w-4" /> Review &amp; Complete
   424→                </Button>
   425→              </div>
   426→            </div>
   427→          </>
   428→        )}
   429→      </DialogContent>
   430→    </Dialog>
   431→  )
   432→}

   437→
   438→export function CompleteTakeDialog({ open, onOpenChange, take, onCompleted }: {
   439→  open: boolean
   440→  onOpenChange: (v: boolean) => void
   441→  take: StockTakeRec | null
   442→  onCompleted: () => void
   443→}) {
   444→  const [applyToStock, setApplyToStock] = useState(true)
   445→  const [note, setNote] = useState('')
   446→  const [submitting, setSubmitting] = useState(false)
   447→  const [varianceLines, setVarianceLines] = useState<StockTakeItemRec[]>([])
   448→  const [loadingLines, setLoadingLines] = useState(false)
   449→
   450→  useEffect(() => {
   451→    if (!open || !take) return
   452→    setApplyToStock(true)
   453→    setNote('')
   454→    setLoadingLines(true)
   455→    fetch(`/api/stock-takes/${take.id}`, { cache: 'no-store' })
   456→      .then((r) => (r.ok ? r.json() : null))
   457→      .then((j) => {
   458→        const lines: StockTakeItemRec[] = j?.take?.items ?? []
   459→        setVarianceLines(lines.filter((i) => (i.varianceQty ?? 0) !== 0))
   460→      })
   461→      .finally(() => setLoadingLines(false))
   462→  }, [open, take])
   463→
   464→  const complete = async () => {
   465→    if (!take) return
   466→    setSubmitting(true)
   467→    try {
   468→      const res = await fetch(`/api/stock-takes/${take.id}/complete`, {
   469→        method: 'PUT',
   470→        headers: { 'Content-Type': 'application/json' },
   471→        body: JSON.stringify({ applyToStock, note: note.trim() || null }),
   472→      })
   473→      const json = await res.json()
   474→      if (!res.ok) {
   475→        toast({ title: 'Cannot complete count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   476→        return
   477→      }
   478→      toast({
   479→        title: `Stock take ${json.take.stockTakeNo} completed`,
   480→        description: applyToStock
   481→          ? `Variances applied to system stock — net ${json.aggregates.varianceQty > 0 ? '+' : ''}${json.aggregates.varianceQty} units.`
   482→          : 'Recorded without changing system stock.',
   483→      })
   484→      onOpenChange(false)
   485→      onCompleted()
   486→    } finally {
   487→      setSubmitting(false)
   488→    }
   489→  }
   490→
   491→  if (!take) return null
   492→  const netValue = take.varianceValue
   493→
   494→  return (
   495→    <Dialog open={open} onOpenChange={onOpenChange}>
   496→      <DialogContent className="max-w-2xl">
   497→        <DialogHeader>
   498→          <DialogTitle className="flex items-center gap-2">

   581→// ---------------------------------------------------------------------------

   583→export function CancelTakeDialog({ open, onOpenChange, take, onCancelled }: {
   584→  open: boolean
   585→  onOpenChange: (v: boolean) => void
   586→  take: StockTakeRec | null
   587→  onCancelled: () => void
   588→}) {
   589→  const [note, setNote] = useState('')
   590→  const [submitting, setSubmitting] = useState(false)
   591→
   592→  useEffect(() => { if (open) setNote('') }, [open])
   593→
   594→  const cancel = async () => {
   595→    if (!take) return
   596→    setSubmitting(true)
   597→    try {
   598→      const res = await fetch(`/api/stock-takes/${take.id}/cancel`, {
   599→        method: 'PUT',
   600→        headers: { 'Content-Type': 'application/json' },
   601→        body: JSON.stringify({ note: note.trim() || null }),
   602→      })
   603→      const json = await res.json()
   604→      if (!res.ok) {
   605→        toast({ title: 'Cannot cancel count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   606→        return
   607→      }
   608→      toast({ title: `Count ${json.take.stockTakeNo} cancelled`, description: 'No stock changes were applied.' })
   609→      onOpenChange(false)
   610→      onCancelled()
   611→    } finally {
   612→      setSubmitting(false)
   613→    }
   614→  }
   615→
   616→  if (!take) return null
   617→  return (
   618→    <Dialog open={open} onOpenChange={onOpenChange}>
   619→      <DialogContent className="sm:max-w-md">
   620→        <DialogHeader>
   621→          <DialogTitle className="flex items-center gap-2">
   622→            <XCircle className="h-5 w-5 text-red-500" /> Cancel {take.stockTakeNo}?
   623→          </DialogTitle>
   624→          <DialogDescription>
   625→            The count session will be kept for audit with status “cancelled”. No stock changes are applied.
   626→          </DialogDescription>
   627→        </DialogHeader>
   628→        <div className="space-y-4">
   629→          <div className="space-y-1.5">
   630→            <Label>Reason (optional)</Label>
   631→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2} placeholder="e.g. delivery arrived mid-count, restarting after hours" />
   632→          </div>
   633→          <div className="flex justify-end gap-2">
   634→            <Button variant="outline" className="rounded-full" onClick={() => onOpenChange(false)}>Keep Counting</Button>
   635→            <Button variant="destructive" className="rounded-full" onClick={cancel} disabled={submitting}>
   636→              {submitting ? 'Cancelling…' : 'Cancel Session'}
   637→            </Button>
   638→          </div>
   639→        </div>
   640→      </DialogContent>
   641→    </Dialog>
   642→  )
   643→}
