    80→export function StartTakeDialog({ open, onOpenChange, onStarted }: {
    81→  open: boolean
    82→  onOpenChange: (v: boolean) => void
    83→  onStarted: (takeId: string) => void
    84→}) {
    85→  const [title, setTitle] = useState('')
    86→  const [scope, setScope] = useState<'all' | 'category'>('all')
    87→  const [categoryId, setCategoryId] = useState('')
    88→  const [note, setNote] = useState('')
    89→  const [categories, setCategories] = useState<Array<{ id: string; name: string; _count: { medicines: number } }>>([])
    90→  const [submitting, setSubmitting] = useState(false)
    91→
    92→  useEffect(() => {
    93→    if (!open) return
    94→    setTitle('')
    95→    setScope('all')
    96→    setCategoryId('')
    97→    setNote('')
    98→    fetch('/api/categories', { cache: 'no-store' })
    99→      .then((r) => (r.ok ? r.json() : { categories: [] }))
   100→      .then((j) => setCategories(j.categories ?? []))
   101→      .catch(() => setCategories([]))
   102→  }, [open])
   103→
   104→  const start = async () => {
   105→    setSubmitting(true)
   106→    try {
   107→      const res = await fetch('/api/stock-takes', {
   108→        method: 'POST',
   109→        headers: { 'Content-Type': 'application/json' },
   110→        body: JSON.stringify({ title: title.trim(), scope, categoryId: scope === 'category' ? categoryId : null, note: note.trim() || null }),
   111→      })
   112→      const json = await res.json()
   113→      if (!res.ok) {
   114→        toast({ title: 'Cannot start stock take', description: json.error ?? 'Unknown error', variant: 'destructive' })
   115→        return
   116→      }
   117→      toast({ title: 'Count session started', description: `${json.take.stockTakeNo} — ${json.aggregates.totalItems} products snapshotted at today's system stock.` })
   118→      onOpenChange(false)
   119→      onStarted(json.take.id)
   120→    } finally {
   121→      setSubmitting(false)
   122→    }
   123→  }
   124→
   125→  return (
   126→    <Dialog open={open} onOpenChange={onOpenChange}>
   127→      <DialogContent className="sm:max-w-md">
   128→        <DialogHeader>
   129→          <DialogTitle className="flex items-center gap-2">
   130→            <PlayCircle className="h-5 w-5 text-teal-600" /> Start Stock Take
   131→          </DialogTitle>
   132→          <DialogDescription>
   133→            A new count session snapshots the current system stock of every product in scope. Enter physical counts against it —
   134→            variances are calculated as counted minus system, positive or negative.
   135→          </DialogDescription>
   136→        </DialogHeader>
   137→        <div className="space-y-4">
   138→          <div className="space-y-1.5">
   139→            <Label>Count title</Label>
   140→            <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Monthly shelf count" aria-label="Count title" />
   141→          </div>
   142→          <div className="space-y-1.5">
   143→            <Label>Scope</Label>
   144→            <Select value={scope} onValueChange={(v) => setScope(v as 'all' | 'category')}>
   145→              <SelectTrigger aria-label="Count scope"><SelectValue /></SelectTrigger>
   146→              <SelectContent>
   147→                <SelectItem value="all">All active products</SelectItem>
   148→                <SelectItem value="category">Single category</SelectItem>
   149→              </SelectContent>
   150→            </Select>
   151→          </div>
   152→          {scope === 'category' && (
   153→            <div className="space-y-1.5">
   154→              <Label>Category to count</Label>
   155→              <Select value={categoryId} onValueChange={setCategoryId}>
   156→                <SelectTrigger aria-label="Category"><SelectValue placeholder="Pick a category" /></SelectTrigger>
   157→                <SelectContent>
   158→                  {categories.map((c) => (
   159→                    <SelectItem key={c.id} value={c.id}>{c.name} ({c._count.medicines})</SelectItem>
   160→                  ))}
   161→                </SelectContent>
   162→              </Select>
   163→            </div>
   164→          )}
   165→          <div className="space-y-1.5">
   166→            <Label>Note (optional)</Label>
   167→            <Input value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. blind count, shelf A–D" />
   168→          </div>
   169→          <Button className="w-full rounded-full bg-teal-600 hover:bg-teal-700" size="lg" onClick={start} disabled={submitting || !title.trim() || (scope === 'category' && !categoryId)}>
   170→            {submitting ? 'Snapshotting stock…' : 'Start Count Session'}
   171→          </Button>
   172→        </div>
   173→      </DialogContent>
   174→    </Dialog>
   175→  )
   176→}

   179→// Pause dialog — park the session; counting resumes any later day

   182→export function PauseTakeDialog({ open, onOpenChange, take, onPaused }: {
   183→  open: boolean
   184→  onOpenChange: (v: boolean) => void
   185→  take: StockTakeRec | null
   186→  onPaused: () => void
   187→}) {
   188→  const [note, setNote] = useState('')
   189→  const [submitting, setSubmitting] = useState(false)
   190→
   191→  useEffect(() => { if (open) setNote('') }, [open])
   192→
   193→  const pause = async () => {
   194→    if (!take) return
   195→    setSubmitting(true)
   196→    try {
   197→      const res = await fetch(`/api/stock-takes/${take.id}/pause`, {
   198→        method: 'PUT',
   199→        headers: { 'Content-Type': 'application/json' },
   200→        body: JSON.stringify({ note: note.trim() || null }),
   201→      })
   202→      const json = await res.json()
   203→      if (!res.ok) {
   204→        toast({ title: 'Cannot pause count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   205→        return
   206→      }
   207→      toast({
   208→        title: `Count ${json.take.stockTakeNo} paused`,
   209→        description: 'Counting is parked — resume any time, even days from now. Every count entered so far is kept.',
   210→      })
   211→      onOpenChange(false)
   212→      onPaused()
   213→    } finally {
   214→      setSubmitting(false)
   215→    }
   216→  }
   217→
   218→  if (!take) return null
   219→  return (
   220→    <Dialog open={open} onOpenChange={onOpenChange}>
   221→      <DialogContent className="sm:max-w-md">
   222→        <DialogHeader>
   223→          <DialogTitle className="flex items-center gap-2">
   224→            <PauseCircle className="h-5 w-5 text-amber-500" /> Pause {take.stockTakeNo}?
   225→          </DialogTitle>
   226→          <DialogDescription>
   227→            The session is parked — nothing is lost. The shop keeps trading normally; when you resume
   228→            (today, tomorrow or next week) the not-yet-counted lines are re-baselined to the then-current
   229→            system stock, and products registered in the meantime join the sheet.
   230→          </DialogDescription>
   231→        </DialogHeader>
   232→        <div className="space-y-4">
   233→          <div className="space-y-1.5">
   234→            <Label>Reason (optional)</Label>
   235→            <Input value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. closing time, counting resumes tomorrow morning" />
   236→          </div>
   237→          <div className="flex justify-end gap-2">
   238→            <Button variant="outline" className="rounded-full" onClick={() => onOpenChange(false)}>Keep Counting</Button>
   239→            <Button className="rounded-full bg-amber-500 hover:bg-amber-600" onClick={pause} disabled={submitting}>
   240→              {submitting ? 'Pausing…' : 'Pause Session'}
   241→            </Button>
   242→          </div>
   243→        </div>
   244→      </DialogContent>
   245→    </Dialog>
   246→  )
   247→}

   253→type LineFilter = 'all' | 'pending' | 'counted' | 'variance'
   254→
   255→export function CountSheetDialog({ open, onOpenChange, takeId, onChanged, onComplete }: {
   256→  open: boolean
   257→  onOpenChange: (v: boolean) => void
   258→  takeId: string
   259→  onChanged: () => void
   260→  onComplete: (take: StockTakeRec) => void
   261→}) {
   262→  const [take, setTake] = useState<StockTakeRec | null>(null)
   263→  const [items, setItems] = useState<StockTakeItemRec[]>([])
   264→  const [loading, setLoading] = useState(true)
   265→  const [search, setSearch] = useState('')
   266→  const [filter, setFilter] = useState<LineFilter>('all')
   267→  const [drafts, setDrafts] = useState<Record<string, string>>({})
   268→  const [savingIds, setSavingIds] = useState<Set<string>>(new Set())
   269→  const [savedIds, setSavedIds] = useState<Set<string>>(new Set())
   270→  const [scannerOpen, setScannerOpen] = useState(false)
   271→  const [focusItemId, setFocusItemId] = useState<string | null>(null)
   272→  const inputRefs = useRef<Map<string, HTMLInputElement>>(new Map())
   273→
   274→  const load = useCallback(async () => {
   275→    try {
   276→      const res = await fetch(`/api/stock-takes/${takeId}`, { cache: 'no-store' })
   277→      if (!res.ok) return
   278→      const json = await res.json()
   279→      setTake(json.take)
   280→      setItems(json.take.items ?? [])
   281→      // Seed drafts with existing counts
   282→      const d: Record<string, string> = {}
   283→      for (const it of json.take.items ?? []) {
   284→        if (it.countedQty !== null) d[it.id] = String(it.countedQty)
   285→      }
   286→      setDrafts(d)
   287→    } finally {
   288→      setLoading(false)
   289→    }
   290→  }, [takeId])
   291→
   292→  useEffect(() => {
   293→    if (!open) return
   294→    setLoading(true)
   295→    setSearch('')
   296→    setFilter('all')
   297→    setSavedIds(new Set())
   298→    setFocusItemId(null)
   299→    load()
   300→  }, [open, load])
   301→
   302→  const saveLine = useCallback(async (item: StockTakeItemRec) => {
   303→    const raw = drafts[item.id]
   304→    if (raw === undefined) return
   305→    const qty = Number(raw)
   306→    if (raw === '' || Number.isNaN(qty) || qty < 0 || !Number.isInteger(qty)) {
   307→      toast({ title: 'Invalid count', description: 'Enter a whole number of 0 or more.', variant: 'destructive' })
   308→      return
   309→    }
   310→    if (item.countedQty === qty) return
   311→    setSavingIds((s) => new Set(s).add(item.id))
   312→    try {
   313→      const res = await fetch(`/api/stock-takes/${takeId}/count`, {
   314→        method: 'PUT',
   315→        headers: { 'Content-Type': 'application/json' },
   316→        body: JSON.stringify({ itemId: item.id, countedQty: qty }),
   317→      })
   318→      const json = await res.json()
   319→      if (!res.ok) {
   320→        toast({ title: 'Could not save count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   321→        return
   322→      }
   323→      setItems((list) => list.map((i) => (i.id === item.id ? { ...i, ...json.item } : i)))
   324→      setTake((t) => (t ? { ...t, ...json.aggregates } : t))
   325→      setSavedIds((s) => new Set(s).add(item.id))
   326→      setSavingIds((s) => { const n = new Set(s); n.delete(item.id); return n })
   327→      onChanged()
   328→    } catch {
   329→      setSavingIds((s) => { const n = new Set(s); n.delete(item.id); return n })
   330→      toast({ title: 'Network error', description: 'Count not saved — try again.', variant: 'destructive' })
   331→    }
   332→  }, [drafts, takeId, onChanged])
   333→
   334→  const filtered = useMemo(() => {
   335→    let list = items
   336→    const q = search.trim().toLowerCase()
   337→    if (q) {
   338→      list = list.filter((i) =>
   339→        i.medicineName.toLowerCase().includes(q) ||
   340→        i.sku.toLowerCase().includes(q) ||
   341→        (i.category ?? '').toLowerCase().includes(q))
   342→    }
   343→    if (filter === 'pending') list = list.filter((i) => i.countedQty === null)
   344→    if (filter === 'counted') list = list.filter((i) => i.countedQty !== null)
   345→    if (filter === 'variance') list = list.filter((i) => (i.varianceQty ?? 0) !== 0)
   346→    return list
   347→  }, [items, search, filter])
   348→
   349→  // Camera scan → resolve the code against the catalog (exact barcode/SKU
   350→  // first, single-hit fallback — same resolution as the POS), then jump to
   351→  // that line in the sheet and focus its count input.
   352→  const resolveScan = useCallback(async (code: string) => {
   353→    try {
   354→      const res = await fetch(`/api/medicines?search=${encodeURIComponent(code)}`, { cache: 'no-store' })
   355→      if (!res.ok) throw new Error('lookup failed')
   356→      const meds: Array<{ id: string; sku: string; barcode: string | null; name: string }> =
   357→        (await res.json()).medicines ?? []
   358→      const q = code.trim().toLowerCase()
   359→      const exact =
   360→        meds.find((m) => m.barcode === code.trim() || m.sku.toLowerCase() === q) ??
   361→        (meds.length === 1 ? meds[0] : undefined)
   362→      if (!exact) {
   363→        toast({ title: 'Product not found', description: `Nothing in the catalog matches “${code}”.`, variant: 'destructive' })
   364→        return
   365→      }
   366→      const line = items.find((i) => i.medicineId === exact.id)
   367→      if (!line) {
   368→        toast({ title: 'Not in this count sheet', description: `${exact.name} (${exact.sku}) is not part of this take.`, variant: 'destructive' })
   369→        return
   370→      }
   371→      setSearch(line.sku)
   372→      setFilter('all')
   373→      setFocusItemId(line.id)
   374→      toast({ title: 'Scanned', description: `${line.medicineName} — enter the counted quantity.` })
   375→    } catch {
   376→      toast({ title: 'Scan failed', description: 'Could not look up that code — try again.', variant: 'destructive' })
   377→    }
   378→  }, [items])
   379→
   380→  // Focus (and select) the count input of the scanned line once it renders.
   381→  useEffect(() => {
   382→    if (!focusItemId) return
   383→    const t = setTimeout(() => {
   384→      const el = inputRefs.current.get(focusItemId)
   385→      el?.focus()
   386→      el?.select()
   387→    }, 80)
   388→    return () => clearTimeout(t)
   389→  }, [focusItemId, search])
   390→
   391→  const progressPct = take && take.totalItems > 0 ? (take.countedItems / take.totalItems) * 100 : 0
   392→  const allCounted = take ? take.countedItems === take.totalItems && take.totalItems > 0 : false
   393→  const paused = take?.status === 'paused'
   394→
   395→  return (
   396→    <Dialog open={open} onOpenChange={onOpenChange}>
   397→      <DialogContent className="max-w-5xl">
   398→        <DialogHeader>
   399→          <DialogTitle className="flex flex-wrap items-center gap-2">
   400→            <PackageSearch className="h-5 w-5 text-teal-600" />
   401→            Physical Count — {take?.stockTakeNo ?? '…'}
   402→            {take && <span className="text-sm font-normal text-muted-foreground">{take.title}</span>}
   403→          </DialogTitle>
   404→          <DialogDescription>
   405→            Type the quantity you physically count on the shelf. Counts save automatically on Enter or when you leave the field.
   406→          </DialogDescription>
   407→        </DialogHeader>
   408→
   409→        {paused && (
   410→          <div className="flex items-center gap-2 rounded-xl border border-amber-200 bg-amber-50/70 px-4 py-3 text-sm text-amber-800 dark:border-amber-500/30 dark:bg-amber-500/10 dark:text-amber-300">
   411→            <PauseCircle className="h-4 w-4 shrink-0" />
   412→            <span>
   413→              This session is paused{take?.pausedAt ? ` since ${fmtDateTime(take.pausedAt)}` : ''} — counts are disabled.
   414→              Resume the session from the Stock Take panel to continue counting.
   415→            </span>
   416→          </div>
   417→        )}
   418→
   419→        {loading ? (
   420→          <div className="space-y-2 py-4">{Array.from({ length: 6 }).map((_, i) => <Skeleton key={i} className="h-10 w-full" />)}</div>
   421→        ) : (
   422→          <>
   423→            {/* Progress + live variance header */}
   424→            <div className="rounded-xl border bg-teal-600/5 p-4">
   425→              <div className="flex flex-wrap items-center justify-between gap-3">
   426→                <div className="min-w-48 flex-1">
   427→                  <div className="flex items-center justify-between text-xs">
   428→                    <span className="font-semibold">{take?.countedItems ?? 0} / {take?.totalItems ?? 0} lines counted</span>
   429→                    <span className="text-muted-foreground">{Math.round(progressPct)}%</span>
   430→                  </div>
   431→                  <Progress value={progressPct} className="mt-2 h-2" aria-label="Count progress" />
   432→                </div>
   433→                <div className="flex flex-wrap items-center gap-2 text-xs">
   434→                  <span className="rounded-full bg-emerald-50 px-2.5 py-1 font-semibold text-emerald-700 dark:bg-emerald-500/10 dark:text-emerald-400">
   435→                    Surplus {take?.positiveLines ?? 0}
   436→                  </span>
   437→                  <span className="rounded-full bg-red-50 px-2.5 py-1 font-semibold text-red-700 dark:bg-red-500/10 dark:text-red-400">
   438→                    Short {take?.negativeLines ?? 0}
   439→                  </span>
   440→                  <span className="rounded-full bg-muted px-2.5 py-1 font-semibold">
   441→                    Net {take ? `${take.varianceQty > 0 ? '+' : ''}${take.varianceQty}` : '0'} units · {take ? `${take.varianceValue >= 0 ? '+' : '−'}${fmtMoney(Math.abs(take.varianceValue)).replace(/^[^0-9]*/, '')}` : ''} value
   442→                  </span>
   443→                </div>
   444→              </div>
   445→            </div>
   446→
   447→            {/* Filters */}
   448→            <div className="flex flex-wrap items-center gap-2">
   449→              <div className="relative flex-1 min-w-48">
   450→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   451→                <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search product, SKU, category…" className="pl-9" aria-label="Search count lines" />
   452→              </div>
   453→              <Button
   454→                size="sm"
   455→                variant="outline"
   456→                className="rounded-full border-teal-500/50 text-teal-700 hover:bg-teal-50 dark:text-teal-300 dark:hover:bg-teal-500/10"
   457→                onClick={() => setScannerOpen(true)}
   458→                title="Scan a barcode with the camera to jump to its line"
   459→              >
   460→                <ScanLine className="h-4 w-4" />
   461→                Scan
   462→              </Button>
   463→              <div className="flex gap-1">
   464→                {([
   465→                  { k: 'all', label: 'All' },
   466→                  { k: 'pending', label: 'Not counted' },
   467→                  { k: 'counted', label: 'Counted' },
   468→                  { k: 'variance', label: 'Variance' },
   469→                ] as Array<{ k: LineFilter; label: string }>).map((f) => (
   470→                  <Button key={f.k} size="sm" variant="outline"
   471→                    className={`rounded-full text-xs ${filter === f.k ? 'border-teal-500 bg-teal-50 text-teal-700 dark:bg-teal-500/10 dark:text-teal-300' : ''}`}
   472→                    onClick={() => setFilter(f.k)}>
   473→                    {f.label}
   474→                  </Button>
   475→                ))}
   476→              </div>
   477→            </div>
   478→
   479→            {/* Lines */}
   480→            <div className="max-h-[46vh] overflow-y-auto rounded-xl border">
   481→              <Table>
   482→                <TableHeader className="sticky top-0 z-10 bg-card">
   483→                  <TableRow className="hover:bg-transparent">
   484→                    <TableHead>Product</TableHead>
   485→                    <TableHead className="text-center">System</TableHead>
   486→                    <TableHead className="w-28 text-center">Counted</TableHead>
   487→                    <TableHead className="text-center">Variance</TableHead>
   488→                    <TableHead className="hidden text-right md:table-cell">Value Impact</TableHead>
   489→                    <TableHead className="w-10"></TableHead>
   490→                  </TableRow>
   491→                </TableHeader>
   492→                <TableBody>
   493→                  {filtered.length === 0 && (

   595→export function CompleteTakeDialog({ open, onOpenChange, take, onCompleted }: {
   596→  open: boolean
   597→  onOpenChange: (v: boolean) => void
   598→  take: StockTakeRec | null
   599→  onCompleted: () => void
   600→}) {
   601→  const [applyToStock, setApplyToStock] = useState(true)
   602→  const [note, setNote] = useState('')
   603→  const [submitting, setSubmitting] = useState(false)
   604→  const [varianceLines, setVarianceLines] = useState<StockTakeItemRec[]>([])
   605→  const [loadingLines, setLoadingLines] = useState(false)
   606→
   607→  useEffect(() => {
   608→    if (!open || !take) return
   609→    setApplyToStock(true)
   610→    setNote('')
   611→    setLoadingLines(true)
   612→    fetch(`/api/stock-takes/${take.id}`, { cache: 'no-store' })
   613→      .then((r) => (r.ok ? r.json() : null))
   614→      .then((j) => {
   615→        const lines: StockTakeItemRec[] = j?.take?.items ?? []
   616→        setVarianceLines(lines.filter((i) => (i.varianceQty ?? 0) !== 0))
   617→      })
   618→      .finally(() => setLoadingLines(false))
   619→  }, [open, take])
   620→
   621→  const complete = async () => {
   622→    if (!take) return
   623→    setSubmitting(true)
   624→    try {
   625→      const res = await fetch(`/api/stock-takes/${take.id}/complete`, {
   626→        method: 'PUT',
   627→        headers: { 'Content-Type': 'application/json' },
   628→        body: JSON.stringify({ applyToStock, note: note.trim() || null }),
   629→      })
   630→      const json = await res.json()
   631→      if (!res.ok) {
   632→        toast({ title: 'Cannot complete count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   633→        return
   634→      }
   635→      toast({
   636→        title: `Stock take ${json.take.stockTakeNo} completed`,
   637→        description: applyToStock
   638→          ? `Variances applied to system stock — net ${json.aggregates.varianceQty > 0 ? '+' : ''}${json.aggregates.varianceQty} units.`
   639→          : 'Recorded without changing system stock.',
   640→      })
   641→      onOpenChange(false)
   642→      onCompleted()
   643→    } finally {
   644→      setSubmitting(false)
   645→    }
   646→  }
   647→
   648→  if (!take) return null
   649→  const netValue = take.varianceValue
   650→
   651→  return (
   652→    <Dialog open={open} onOpenChange={onOpenChange}>
   653→      <DialogContent className="max-w-2xl">
   654→        <DialogHeader>
   655→          <DialogTitle className="flex items-center gap-2">
   656→            <ClipboardCheck className="h-5 w-5 text-teal-600" /> Complete {take.stockTakeNo}
   657→          </DialogTitle>
   658→          <DialogDescription>Review the variances found during this count before closing the session.</DialogDescription>
   659→        </DialogHeader>
   660→
   661→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   662→          {[
   663→            { label: 'Lines Counted', value: `${take.countedItems}/${take.totalItems}` },
   664→            { label: 'Surplus Lines', value: `+${take.positiveLines}`, cls: 'text-emerald-600 dark:text-emerald-400' },
   665→            { label: 'Shortage Lines', value: `${take.negativeLines}`, cls: 'text-red-600 dark:text-red-400' },
   666→            { label: 'Net Units', value: `${take.varianceQty > 0 ? '+' : ''}${take.varianceQty}` },
   667→          ].map((s) => (
   668→            <div key={s.label} className="rounded-xl border bg-card p-3 text-center">
   669→              <p className="text-[11px] text-muted-foreground">{s.label}</p>
   670→              <p className={`text-lg font-bold tabular-nums ${s.cls ?? ''}`}>{s.value}</p>
   671→            </div>
   672→          ))}
   673→        </div>
   674→
   675→        <div className={`rounded-xl border p-3 text-center ${netValue >= 0 ? 'border-emerald-200 bg-emerald-50/60 dark:border-emerald-500/20 dark:bg-emerald-500/5' : 'border-red-200 bg-red-50/60 dark:border-red-500/20 dark:bg-red-500/5'}`}>
   676→          <p className="text-xs text-muted-foreground">Net stock value variance (counted − system, at cost)</p>
   677→          <p className={`text-2xl font-bold tabular-nums ${netValue >= 0 ? 'text-emerald-700 dark:text-emerald-400' : 'text-red-700 dark:text-red-400'}`}>
   678→            {netValue >= 0 ? '+' : '−'}{fmtMoney(Math.abs(netValue))}
   679→          </p>
   680→        </div>
   681→
   682→        {/* Variance lines */}
   683→        <div className="max-h-52 overflow-y-auto rounded-xl border">
   684→          {loadingLines ? (
   685→            <div className="space-y-2 p-3">{Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-8 w-full" />)}</div>
   686→          ) : varianceLines.length === 0 ? (
   687→            <p className="flex items-center justify-center gap-2 p-6 text-sm text-muted-foreground">
   688→              <CheckCircle2 className="h-4 w-4 text-emerald-500" /> Perfect count — every line matched the system stock.
   689→            </p>
   690→          ) : (
   691→            <Table>
   692→              <TableHeader>
   693→                <TableRow className="hover:bg-transparent">
   694→                  <TableHead>Product</TableHead>
   695→                  <TableHead className="text-center">System</TableHead>
   696→                  <TableHead className="text-center">Counted</TableHead>
   697→                  <TableHead className="text-center">Variance</TableHead>
   698→                  <TableHead className="text-right">Value</TableHead>
   699→                </TableRow>
   700→              </TableHeader>
   701→              <TableBody>
   702→                {varianceLines.map((i) => (
   703→                  <TableRow key={i.id}>
   704→                    <TableCell className="max-w-48 truncate text-sm font-medium">{i.medicineName}</TableCell>
   705→                    <TableCell className="text-center tabular-nums">{i.systemQty}</TableCell>
   706→                    <TableCell className="text-center tabular-nums">{i.countedQty}</TableCell>
   707→                    <TableCell className="text-center"><VarianceBadge variance={i.varianceQty ?? 0} /></TableCell>
   708→                    <TableCell className="text-right"><VarianceValueBadge value={i.varianceValue ?? 0} /></TableCell>
   709→                  </TableRow>
   710→                ))}
   711→              </TableBody>
   712→            </Table>
   713→          )}
   714→        </div>
   715→
   716→        <div className="flex items-start justify-between gap-4 rounded-xl border p-3">
   717→          <div className="space-y-0.5">
   718→            <Label htmlFor="apply-stock" className="font-medium">Apply variances to system stock</Label>
   719→            <p className="text-xs text-muted-foreground">Adjusts stock by the variance delta and writes an audit log per product. Turn off to record the count without touching stock.</p>
   720→          </div>
   721→          <Switch id="apply-stock" checked={applyToStock} onCheckedChange={setApplyToStock} />
   722→        </div>
   723→        <div className="space-y-1.5">
   724→          <Label>Completion note (optional)</Label>
   725→          <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2} placeholder="e.g. variances reviewed with manager, breakage confirmed" />
   726→        </div>
   727→
   728→        <Button className="w-full rounded-full bg-teal-600 hover:bg-teal-700" size="lg" onClick={complete} disabled={submitting}>
   729→          {submitting ? 'Completing…' : 'Complete Stock Take'}
   730→        </Button>
   731→      </DialogContent>
   732→    </Dialog>
   733→  )
   734→}

   740→export function CancelTakeDialog({ open, onOpenChange, take, onCancelled }: {
   741→  open: boolean
   742→  onOpenChange: (v: boolean) => void
   743→  take: StockTakeRec | null
   744→  onCancelled: () => void
   745→}) {
   746→  const [note, setNote] = useState('')
   747→  const [submitting, setSubmitting] = useState(false)
   748→
   749→  useEffect(() => { if (open) setNote('') }, [open])
   750→
   751→  const cancel = async () => {
   752→    if (!take) return
   753→    setSubmitting(true)
   754→    try {
   755→      const res = await fetch(`/api/stock-takes/${take.id}/cancel`, {
   756→        method: 'PUT',
   757→        headers: { 'Content-Type': 'application/json' },
   758→        body: JSON.stringify({ note: note.trim() || null }),
   759→      })
   760→      const json = await res.json()
   761→      if (!res.ok) {
   762→        toast({ title: 'Cannot cancel count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   763→        return
   764→      }
   765→      toast({ title: `Count ${json.take.stockTakeNo} cancelled`, description: 'No stock changes were applied.' })
   766→      onOpenChange(false)
   767→      onCancelled()
   768→    } finally {
   769→      setSubmitting(false)
   770→    }
   771→  }
   772→
   773→  if (!take) return null
   774→  return (
   775→    <Dialog open={open} onOpenChange={onOpenChange}>
   776→      <DialogContent className="sm:max-w-md">
   777→        <DialogHeader>
   778→          <DialogTitle className="flex items-center gap-2">
   779→            <XCircle className="h-5 w-5 text-red-500" /> Cancel {take.stockTakeNo}?
   780→          </DialogTitle>
   781→          <DialogDescription>
   782→            The count session will be kept for audit with status “cancelled”. No stock changes are applied.
   783→          </DialogDescription>
   784→        </DialogHeader>
   785→        <div className="space-y-4">
   786→          <div className="space-y-1.5">
   787→            <Label>Reason (optional)</Label>
   788→            <Textarea value={note} onChange={(e) => setNote(e.target.value)} rows={2} placeholder="e.g. delivery arrived mid-count, restarting after hours" />
   789→          </div>
   790→          <div className="flex justify-end gap-2">
   791→            <Button variant="outline" className="rounded-full" onClick={() => onOpenChange(false)}>Keep Counting</Button>

   793→              {submitting ? 'Cancelling…' : 'Cancel Session'}
   794→            </Button>
   795→          </div>
   796→        </div>
   797→      </DialogContent>
   798→    </Dialog>
   799→  )
   800→}
