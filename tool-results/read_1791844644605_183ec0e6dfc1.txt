    89→  // Checkout state

    94→
    95→  // Discount approval workflow state
    96→  const [approvalOpen, setApprovalOpen] = useState(false)
    97→  const [approvedRequests, setApprovedRequests] = useState<ApprovedDiscountRequest[]>([])
    98→  const [approvedListOpen, setApprovedListOpen] = useState(false)
    99→  const [discountRequestId, setDiscountRequestId] = useState<string | null>(null)
   100→
   101→  const searchRef = useRef<HTMLInputElement>(null)
   102→

   106→        fetch('/api/medicines?status=active', { cache: 'no-store' }),

   118→

   121→  const loadCurrentShift = useCallback(async () => {

   125→      const json = await res.json()

   128→      /* non-fatal */
   129→    }

   131→
   132→  useEffect(() => { loadCurrentShift() }, [loadCurrentShift])
   133→

   137→      const res = await fetch('/api/discount-requests?status=approved&mine=1', { cache: 'no-store' })
   138→      if (!res.ok) return
   139→      const json = await res.json()
   140→      setApprovedRequests(json.approvedForMe ?? [])

   142→      /* non-fatal */
   143→    }
   144→  }, [])
   145→
   146→  useEffect(() => { loadApprovedRequests() }, [loadApprovedRequests])
   147→
   148→  useEffect(() => {
   149→    if (quickSearch) setSearch(quickSearch)
   150→  }, [quickSearch])
   151→
   152→  const filtered = useMemo(() => {
   153→    let list = meds
   154→    if (activeCat !== 'all') list = list.filter((m) => m.categoryId === activeCat)
   155→    if (search.trim()) {
   156→      const q = search.trim().toLowerCase()
   157→      list = list.filter((m) => m.name.toLowerCase().includes(q) || m.sku.toLowerCase().includes(q) || (m.barcode ?? '').includes(q) || (m.genericName ?? '').toLowerCase().includes(q) || (m.brand ?? '').toLowerCase().includes(q))
   158→    }
   159→    return list
   160→  }, [meds, activeCat, search])
   161→
   162→  const addToCart = useCallback((m: Medicine, qty = 1) => {
   163→    if (m.stockQty <= 0) {
   164→      toast({ title: 'Out of stock', description: `${m.name} has no stock available.`, variant: 'destructive' })
   165→      return
   166→    }
   167→    setCart((prev) => {
   168→      const existing = prev.find((l) => l.medicineId === m.id)
   169→      if (existing) {
   170→        if (existing.qty + qty > m.stockQty) {
   171→          toast({ title: 'Stock limit reached', description: `Only ${m.stockQty} ${m.unit}(s) of ${m.name} available.`, variant: 'destructive' })
   172→          return prev
   173→        }
   174→        return prev.map((l) => (l.medicineId === m.id ? { ...l, qty: l.qty + qty } : l))
   175→      }
   176→      return [
   177→        ...prev,
   178→        { medicineId: m.id, name: m.name, sku: m.sku, unit: m.unit, unitPrice: m.effectiveSellingPrice ?? m.sellingPrice, qty, stockQty: m.stockQty, prescriptionRequired: m.prescriptionRequired },
   179→      ]
   180→    })

   182→
   183→  // Barcode scanner behavior: Enter in search adds exact match or first result
   184→  const onSearchKeyDown = (e: React.KeyboardEvent<HTMLInputElement>) => {
   185→    if (e.key === 'Enter' && filtered.length > 0) {
   186→      const exactBarcode = meds.find((m) => m.barcode === search.trim() || m.sku.toLowerCase() === search.trim().toLowerCase())
   187→      addToCart(exactBarcode ?? filtered[0])
   188→      setSearch('')
   189→      searchRef.current?.focus()
   190→    }
   191→  }
   192→
   193→  const updateQty = (id: string, delta: number) => {
   194→    setCart((prev) =>
   195→      prev
   196→        .map((l) => {
   197→          if (l.medicineId !== id) return l
   198→          const newQty = l.qty + delta
   199→          if (newQty > l.stockQty) {
   200→            toast({ title: 'Stock limit reached', description: `Only ${l.stockQty} ${l.unit}(s) available.`, variant: 'destructive' })
   201→            return l
   202→          }
   203→          return { ...l, qty: newQty }
   204→        })
   205→        .filter((l) => l.qty > 0)
   206→    )
   207→  }
   208→
   209→  const removeLine = (id: string) => setCart((prev) => prev.filter((l) => l.medicineId !== id))
   210→
   211→  const subtotal = useMemo(() => +cart.reduce((s, l) => s + l.unitPrice * l.qty, 0).toFixed(2), [cart])
   212→  const discountValue = useMemo(() => Math.min(Number(orderDiscount) || 0, subtotal), [orderDiscount, subtotal])
   213→  const taxValue = useMemo(() => +(((subtotal - discountValue) * 2.5) / 100).toFixed(2), [subtotal, discountValue])
   214→  const total = useMemo(() => +(subtotal - discountValue + taxValue).toFixed(2), [subtotal, discountValue, taxValue])
   215→  const rxInCart = cart.some((l) => l.prescriptionRequired)
   216→  const tenderedNum = Number(tendered) || 0
   217→  const change = Math.max(0, +(tenderedNum - total).toFixed(2))
   218→
   219→  const rxCount = cart.filter((l) => l.prescriptionRequired).length
   220→
   221→  const policy = useMemo(() => discountPolicyFromSettings(storeSettings ?? null), [storeSettings])
   222→  const needsApproval = requiresDiscountApproval(policy, subtotal, discountValue)
   223→
   224→  const completeSale = async (extra?: { approval?: { username: string; password: string }; discountRequestId?: string | null }): Promise<boolean> => {
   225→    setSubmitting(true)
   226→    try {
   227→      const res = await fetch('/api/sales', {
   228→        method: 'POST',
   229→        headers: { 'Content-Type': 'application/json' },
   230→        body: JSON.stringify({
   231→          items: cart.map((l) => ({ medicineId: l.medicineId, qty: l.qty, unitPrice: l.unitPrice })),
   232→          discount: discountValue,
   233→          customerId: customerId === 'walkin' ? null : customerId,
   234→          paymentMethod,
   235→          paidAmount: paymentMethod === 'cash' ? tenderedNum : total,
   236→          taxRate: 2.5,
   237→          note: note || null,
   238→          ...(extra?.approval ? { approval: extra.approval } : {}),
   239→          ...(extra?.discountRequestId ? { discountRequestId: extra.discountRequestId } : discountRequestId ? { discountRequestId } : {}),
   240→        }),
   241→      })
   242→      const json = await res.json()
   243→      if (!res.ok) {
   244→        if (json.requiresApproval) {
   245→          setApprovalOpen(true)
   246→        } else {
   247→          toast({ title: 'Sale failed', description: json.error ?? 'Unknown error', variant: 'destructive' })
   248→        }
   249→        return false
   250→      }
   251→      setReceipt({
   252→        ...json.sale,
   253→        customerName: json.sale?.customer?.name ?? null,
   254→        shiftNo: json.sale?.shift?.shiftNo ?? null,
   255→        approvedByName: json.sale?.approvedByName ?? null,
   256→        branchName: json.sale?.branchRef?.name ?? null,
   257→      })
   258→      setCart([])
   259→      setOrderDiscount(0)
   260→      setTendered('')
   261→      setNote('')
   262→      setCustomerId('walkin')
   263→      setDiscountRequestId(null)
   264→      setCheckoutOpen(false)
   265→      onSalesRecorded?.()
   266→      loadProducts()
   267→      loadCurrentShift()
   268→      return true
   269→    } finally {
   270→      setSubmitting(false)
   271→    }
   272→  }
   273→
   274→  const checkout = async () => {
   275→    if (paymentMethod === 'cash' && tenderedNum < total) {
   276→      toast({ title: 'Insufficient cash', description: `Tendered ${fmtMoney(tenderedNum)} is less than total ${fmtMoney(total)}.`, variant: 'destructive' })
   277→      return
   278→    }
   279→    await completeSale()
   280→  }
   281→
   282→  const authorizeAndComplete = async (approval: { username: string; password: string }): Promise<boolean> => {
   283→    if (paymentMethod === 'cash' && tenderedNum < total) {
   284→      toast({ title: 'Insufficient cash', description: `Tendered ${fmtMoney(tenderedNum)} is less than total ${fmtMoney(total)}.`, variant: 'destructive' })
   285→      return false
   286→    }
   287→    return await completeSale({ approval })
   288→  }
   289→
   290→  const loadApprovedRequest = (req: ApprovedDiscountRequest) => {
   291→    // Rebuild the cart from the approved snapshot (server re-validates stock)
   292→    setCart(
   293→      req.items.map((l) => ({
   294→        medicineId: l.medicineId,
   295→        name: l.name,
   296→        sku: l.sku,
   297→        unit: l.unit || 'Unit',
   298→        unitPrice: l.unitPrice,
   299→        qty: l.qty,
   300→        stockQty: Number.MAX_SAFE_INTEGER, // validated server-side
   301→        prescriptionRequired: false,
   302→      })),
   303→    )
   304→    setOrderDiscount(req.discountValue)
   305→    setDiscountRequestId(req.id)
   306→    setApprovedListOpen(false)
   307→    toast({ title: `${req.requestCode} loaded`, description: `Discount ${fmtMoney(req.discountValue)} approved by ${req.reviewedByName ?? 'manager'} — complete the sale as usual.` })
   308→  }
   309→
   310→  const methodIcon = (m: string) => (m === 'cash' ? <Banknote className="h-4 w-4" /> : m === 'card' ? <CreditCard className="h-4 w-4" /> : m === 'mobile_money' ? <Smartphone className="h-4 w-4" /> : <ShieldCheck className="h-4 w-4" />)
   311→
   312→  return (
   313→    <div className="grid gap-4 xl:grid-cols-[1fr_400px]">
   314→      {/* Products */}
   315→      <div className="min-w-0 space-y-3">
   316→        {!shift && (
   317→          <div className="flex flex-col gap-3 rounded-xl border border-amber-300 bg-amber-50 px-4 py-3 sm:flex-row sm:items-center sm:justify-between dark:border-amber-500/30 dark:bg-amber-500/10">
   318→            <div className="flex items-center gap-3">
   319→              <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg bg-amber-500/15 text-amber-600 dark:text-amber-400">
   320→                <TriangleAlert className="h-4.5 w-4.5" />
   321→              </span>
   322→              <div>
   323→                <p className="text-sm font-semibold text-amber-800 dark:text-amber-200">No active shift — sales are not being recorded</p>
   324→                <p className="text-xs text-amber-700/80 dark:text-amber-300/80">Start your shift to open the cash drawer and track sales, quantities and cash on this register.</p>
   325→              </div>
   326→            </div>
   327→            <Button size="sm" className="shrink-0 rounded-full bg-amber-500 hover:bg-amber-600 text-white" onClick={() => setStartShiftOpen(true)}>
   328→              <Timer className="h-4 w-4" /> Start Shift
   329→            </Button>
   330→          </div>
   331→        )}
   332→        {shift && (
   333→          <div className="flex flex-wrap items-center gap-2 rounded-xl border border-teal-600/20 bg-teal-600/5 px-4 py-2.5 text-sm dark:bg-teal-500/10">
   334→            <span className="relative flex h-2 w-2">
   335→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-teal-500 opacity-75" />
   336→              <span className="relative inline-flex h-2 w-2 rounded-full bg-teal-500" />
   337→            </span>
   338→            <Timer className="h-4 w-4 text-teal-600 dark:text-teal-400" />
   339→            <span className="font-semibold">{shift.shiftNo}</span>
   340→            <span className="text-xs text-muted-foreground">active shift — sales are being recorded</span>
   341→          </div>
   342→        )}
   343→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   344→          <div className="relative flex-1">
   345→            <Barcode className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   346→            <Input
   347→              ref={searchRef}
   348→              value={search}
   349→              onChange={(e) => setSearch(e.target.value)}
   350→              onKeyDown={onSearchKeyDown}
   351→              placeholder="Scan barcode or search by name / generic / brand… (Enter to add)"
   352→              className="pl-9"
   353→              aria-label="Product search"
   354→            />
   355→            {search && (
   356→              <button onClick={() => setSearch('')} className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground" aria-label="Clear search">
   357→                <X className="h-4 w-4" />
   358→              </button>
   359→            )}
   360→          </div>
   361→          <Badge variant="outline" className="hidden shrink-0 rounded-full px-3 py-1.5 text-xs sm:inline-flex">
   362→            {filtered.length} products
   363→          </Badge>
   364→        </div>
   365→
   366→        <Tabs value={activeCat} onValueChange={setActiveCat}>
   367→          <ScrollArea className="w-full whitespace-nowrap pb-1">
   368→            <TabsList className="inline-flex h-9">
   369→              <TabsTrigger value="all" className="rounded-full px-4 text-xs">All</TabsTrigger>
   370→              {categories.map((c) => (
   371→                <TabsTrigger key={c.id} value={c.id} className="rounded-full px-4 text-xs">{c.name}</TabsTrigger>
   372→              ))}
   373→            </TabsList>
   374→          </ScrollArea>
   375→        </Tabs>
   376→
   377→        {loading ? (
   378→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 2xl:grid-cols-4">
   379→            {Array.from({ length: 8 }).map((_, i) => <Card key={i} className="h-36 animate-pulse bg-muted" />)}

   381→        ) : (
   382→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 2xl:grid-cols-4">

   384→              const out = m.stockQty <= 0
   385→              const low = m.stockQty > 0 && m.stockQty <= 20

   388→                  key={m.id}
   389→                  onClick={() => addToCart(m)}

   391→                  className={`group relative flex flex-col rounded-xl border bg-card p-3.5 text-left transition ${out ? 'cursor-not-allowed opacity-50' : 'hover:-translate-y-0.5 hover:border-teal-400 hover:shadow-lg dark:hover:border-teal-700'}`}
   392→                >
   393→                  <div className="flex items-start justify-between gap-2">
   394→                    <span className="flex h-9 w-9 items-center justify-center rounded-lg bg-teal-600/10 text-teal-600 dark:text-teal-400">
   395→                      <Pill className="h-4.5 w-4.5" />
   396→                    </span>
   397→                    {m.prescriptionRequired && (
   398→                      <Badge className="rounded-full bg-sky-100 text-[10px] font-bold text-sky-700 dark:bg-sky-500/15 dark:text-sky-400" variant="secondary">Rx</Badge>
   399→                    )}
   400→                  </div>
   401→                  <p className="mt-2.5 line-clamp-2 min-h-9 text-sm font-semibold leading-snug">{m.name}</p>
   402→                  <p className="mt-0.5 truncate text-[11px] text-muted-foreground">{m.genericName ?? m.brand ?? m.sku}</p>
   403→                  <div className="mt-2 flex items-end justify-between">
   404→                    <div>
   405→                      <p className="text-base font-bold text-teal-600 dark:text-teal-400">{fmtMoney(m.effectiveSellingPrice ?? m.sellingPrice)}</p>
   406→                      <p className={`text-[11px] font-medium ${out ? 'text-red-500' : low ? 'text-amber-600 dark:text-amber-400' : 'text-muted-foreground'}`}>
   407→                        {out ? 'Out of stock' : `${m.stockQty} ${m.unit}`}
   408→                      </p>
   409→                    </div>
   410→                    {!out && (
   411→                      <span className="flex h-7 w-7 items-center justify-center rounded-full bg-teal-600 text-white opacity-0 shadow transition group-hover:opacity-100">
   412→                        <Plus className="h-4 w-4" />
   413→                      </span>
   414→                    )}
   415→                  </div>
   416→                </button>
   417→              )
   418→            })}
   419→            {filtered.length === 0 && (
   420→              <div className="col-span-full flex flex-col items-center gap-2 py-16 text-muted-foreground">
   421→                <Search className="h-8 w-8 opacity-40" />
   422→                <p className="text-sm">No products match your search.</p>
   423→              </div>
   424→            )}
   425→          </div>
   426→        )}
   427→      </div>
   428→
   429→      {/* Cart */}
   430→      <Card className="flex h-fit flex-col xl:sticky xl:top-20 xl:max-h-[calc(100vh-6rem)]">
   431→        <div className="flex items-center justify-between border-b px-4 py-3">
   432→          <div className="flex items-center gap-2">
   433→            <ShoppingCart className="h-4.5 w-4.5 text-teal-600 dark:text-teal-400" />
   434→            <h3 className="font-semibold">Current Sale</h3>
   435→            {cart.length > 0 && <Badge className="rounded-full bg-teal-600 text-white">{cart.reduce((n, l) => n + l.qty, 0)}</Badge>}
   436→          </div>
   437→          {shift ? (
   438→            <Badge variant="outline" className="gap-1 rounded-full border-teal-300 bg-teal-50 text-[10px] text-teal-700 dark:border-teal-500/30 dark:bg-teal-500/10 dark:text-teal-300">
   439→              <Timer className="h-3 w-3" /> {shift.shiftNo}
   440→            </Badge>
   441→          ) : (
   442→            <button onClick={() => setStartShiftOpen(true)} className="text-[11px] font-medium text-amber-600 underline-offset-2 hover:underline dark:text-amber-400">
   443→              start shift
   444→            </button>
   445→          )}
   446→          {cart.length > 0 && (
   447→            <Button variant="ghost" size="sm" onClick={() => { setCart([]); setOrderDiscount(0); setDiscountRequestId(null) }} className="h-8 gap-1 text-xs text-red-500 hover:text-red-600">
   448→              <Trash2 className="h-3.5 w-3.5" /> Clear
   449→            </Button>
   450→          )}
   451→        </div>
   452→
   453→        {/* Approved discount requests ready to load */}
