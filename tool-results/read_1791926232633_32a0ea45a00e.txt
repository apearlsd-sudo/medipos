     1→// ---------------------------------------------------------------------------
     2→// MediPOS Pro — sync op applier
     3→//
     4→// Turns a wire operation into a local database mutation. Used on BOTH sides:
     5→//   • branch pulling HQ ops  (asHQ: false — HQ takes precedence on conflicts)
     6→//   • HQ applying branch ops (asHQ: true — replicated as-is, prefixed keys)
     7→//
     8→// Ops are idempotent where it matters: sales are keyed by their origin id,
     9→// ledger/loyalty entries by the origin entry id, so a replay can never
    10→// duplicate stock or money movements.
    11→// ---------------------------------------------------------------------------
    12→
    13→import { db } from '@/lib/db'
    14→import { publishLive } from '@/lib/live'
    15→import { planPackSale } from '@/lib/packs'
    16→import { deriveSkuForBranch } from '@/lib/stock-transfer'
    17→
    18→export interface WireOp {
    19→  id: string
    20→  nodeId: string
    21→  seq: number
    22→  op: string
    23→  targetKey?: string | null
    24→  payload: Record<string, unknown> & { id?: string }
    25→  createdAt: string
    26→}
    27→
    28→export type ApplyResult = 'ok' | 'duplicate' | 'skipped'
    29→
    30→const str = (v: unknown, fallback = ''): string => (typeof v === 'string' ? v : fallback)
    31→const num = (v: unknown, fallback = 0): number => (typeof v === 'number' && Number.isFinite(v) ? v : fallback)
    32→const date = (v: unknown): Date | null => {
    33→  if (typeof v !== 'string') return null
    34→  const d = new Date(v)
    35→  return Number.isNaN(d.getTime()) ? null : d
    36→}
    37→
    38→export async function applySyncOp(
    39→  op: WireOp,
    40→  opts: { asHQ: boolean; peerName?: string },
    41→): Promise<ApplyResult> {
    42→  const p = op.payload ?? {}
    43→  switch (op.op) {
    44→    case 'sale.created':
    45→      return applySaleCreated(op, p, opts)
    46→    case 'sale.verified':
    47→      return applySaleVerified(op, p)
    48→    case 'medicine.updated':
    49→    case 'medicine.created':
    50→      return applyMedicineUpsert(op, p, opts)
    51→    case 'medicine.upsert':
    52→      return applyCatalogUpsertBySku(op, p, opts)
    53→    case 'stock.delta':
    54→    case 'stock.set':
    55→      return applyStockOpBySku(op, p)
    56→    case 'inventory.adjusted':
    57→      return applyInventoryAdjusted(op, p)
    58→    case 'customer.updated':
    59→      return applyCustomerUpdated(op, p)
    60→    case 'ledger.entry':
    61→      return applyLedgerEntry(op, p)
    62→    case 'loyalty.entry':
    63→      return applyLoyaltyEntry(op, p)
    64→    case 'claim.status':
    65→      return applyClaimStatus(op, p)
    66→    case 'insurer.updated':
    67→      return applyInsurerUpsert(op, p)
    68→    case 'stock.returned':
    69→      return applyStockReturned(op, p)
    70→    case 'expense.created':
    71→      return applyExpenseCreated(op, p, opts)
    72→    case 'expense.voided':
    73→      return applyExpenseVoided(op, p)
    74→    case 'quotation.updated':
    75→      return applyQuotationUpdated(op, p, opts)
    76→    case 'quotation.status':
    77→      return applyQuotationStatus(op, p)
    78→    default:
    79→      return 'skipped'
    80→  }
    81→}

    85→async function applySaleCreated(
    86→  op: WireOp,
    87→  p: Record<string, unknown>,
    88→  opts: { asHQ: boolean; peerName?: string },
    89→): Promise<ApplyResult> {
    90→  const originId = str(p.id) || op.id
    91→  const existing = await db.sale.findUnique({ where: { id: originId } })
    92→  if (existing) return 'duplicate'
    93→
    94→  const localInvoice = opts.asHQ && op.nodeId !== 'HQ'
    95→    ? `${op.nodeId}:${str(p.invoiceNo)}`
    96→    : str(p.invoiceNo)
    97→
    98→  // Stock at the receiving node mirrors the origin's decrement only when the
    99→  // medicine exists here AND it is the same physical stock (same branch id).
   100→  const items = Array.isArray(p.items) ? p.items : []
   101→  await db.$transaction(async (tx) => {
   102→    await tx.sale.create({
   103→      data: {
   104→        id: originId,
   105→        invoiceNo: localInvoice,
   106→        customerId: str(p.customerId) || null,
   107→        branchId: str(p.branchId) || null,
   108→        shiftId: null,
   109→        subtotal: num(p.subtotal),
   110→        discount: num(p.discount),
   111→        tax: num(p.tax),
   112→        total: num(p.total),
   113→        paidAmount: num(p.paidAmount),
   114→        changeAmount: num(p.changeAmount),
   115→        paymentMethod: str(p.paymentMethod, 'cash'),
   116→        paymentProvider: str(p.paymentProvider) || null,
   117→        paymentReference: str(p.paymentReference) || null,
   118→        paymentStatus: str(p.paymentStatus, 'paid'),
   119→        pointsEarned: num(p.pointsEarned),
   120→        pointsRedeemed: num(p.pointsRedeemed),
   121→        pointsValue: num(p.pointsValue),
   122→        insuranceAmount: num(p.insuranceAmount),
   123→        saleStatus: str(p.saleStatus, 'completed'),
   124→        cashierName: str(p.cashierName, opts.peerName ?? 'Sync'),
   125→        note: str(p.note) || null,
   126→        clientRef: `sync:${originId}`,
   127→        items: {
   128→          create: items.map((it) => {
   129→            const item = it as Record<string, unknown>
   130→            return {
   131→              medicineId: str(item.medicineId) || null,
   132→              name: str(item.name, 'Item'),
   133→              sku: str(item.sku) || null,
   134→              qty: num(item.qty, 1),
   135→              unitPrice: num(item.unitPrice),
   136→              discount: num(item.discount),
   137→              total: num(item.total),
   138→            }
   139→          }),
   140→        },
   141→      },
   142→    })
   143→    // Mirror the stock decrement for items that exist locally (HQ keeps a
   144→    // shadow of branch stock so dashboards can show network-wide levels).
   145→    // Pack-aware: inner-unit sales break packs via the same math as the origin.
   146→    const mirrors = new Map<string, { packs: number; looseAfter: number; innerTotal: number }>()
   147→    for (const it of items) {
   148→      const item = it as Record<string, unknown>
   149→      const mid = str(item.medicineId)
   150→      const qty = num(item.qty, 0)
   151→      if (!mid || qty <= 0) continue
   152→      const med = await tx.medicine.findUnique({ where: { id: mid } })
   153→      if (!med) continue
   154→      // Only decrement when this node actually owns the stock (same branch
   155→      // scope or HQ mirroring a branch sale into its own catalog copy).
   156→      if (med.branchId && str(p.branchId) && med.branchId !== str(p.branchId)) continue
   157→      const saleUnit = str(item.saleUnit) === 'inner' ? 'inner' : 'pack'
   158→      const plan = planPackSale({ stockQty: med.stockQty, looseUnits: med.looseUnits, innerQty: med.innerQty ?? 1 }, saleUnit, qty)
   159→      if (!plan.ok) continue // mirror stock exhausted — the mirror can't go negative
   160→      const d = mirrors.get(mid) ?? { packs: 0, looseAfter: plan.looseAfter, innerTotal: 0 }
   161→      d.packs += plan.packs
   162→      d.looseAfter = plan.looseAfter
   163→      d.innerTotal += num(item.innerTotal, plan.innerTotal) || plan.innerTotal
   164→      mirrors.set(mid, d)
   165→    }
   166→    for (const [mid, d] of mirrors) {
   167→      await tx.medicine.update({ where: { id: mid }, data: { stockQty: { decrement: d.packs }, looseUnits: d.looseAfter } })
   168→      await tx.inventoryLog.create({
   169→        data: { medicineId: mid, type: 'sale_out', quantity: -d.innerTotal, reason: `Synced sale ${localInvoice}`, reference: localInvoice },
   170→      })
   171→    }
   172→  })
   173→  publishLive(['sync', 'sale'], `Sale ${localInvoice} synced from ${opts.peerName ?? op.nodeId}`, 'sync')
   174→  return 'ok'
   175→}

   179→async function applySaleVerified(op: WireOp, p: Record<string, unknown>): Promise<ApplyResult> {
   180→  const id = str(p.id)
   181→  if (!id) return 'skipped'
   182→  const sale = await db.sale.findUnique({ where: { id } })
   183→  if (!sale) return 'skipped'
   184→  const status = str(p.status, 'paid')
   185→  if (sale.paymentStatus === status) return 'duplicate'
   186→  await db.sale.update({
   187→    where: { id },
   188→    data: {
   189→      paymentStatus: status,
   190→      paidAmount: status === 'paid' ? +(sale.total - sale.insuranceAmount - sale.pointsValue).toFixed(2) : sale.paidAmount,
   191→    },
   192→  })
   193→  return 'ok'
   194→}

   196→async function applyMedicineUpsert(
   197→  op: WireOp,
   198→  p: Record<string, unknown>,
   199→  opts: { asHQ: boolean },
   200→): Promise<ApplyResult> {
   201→  const id = str(p.id) || op.targetKey || ''
   202→  if (!id) return 'skipped'
   203→  const existing = await db.medicine.findUnique({ where: { id } })
   204→  const data = {
   205→    name: str(p.name, existing?.name ?? 'Unnamed'),
   206→    genericName: str(p.genericName) || null,
   207→    brand: str(p.brand) || null,
   208→    categoryId: str(p.categoryId, existing?.categoryId ?? ''),
   209→    supplierId: str(p.supplierId) || null,
   210→    unit: str(p.unit, existing?.unit ?? 'Box'),
   211→    batchNumber: str(p.batchNumber) || null,
   212→    expiryDate: date(p.expiryDate),
   213→    costPrice: num(p.costPrice, existing?.costPrice ?? 0),
   214→    sellingPrice: num(p.sellingPrice, existing?.sellingPrice ?? 0),
   215→    prescriptionRequired: p.prescriptionRequired === true,
   216→    insuranceCovered: p.insuranceCovered === true,
   217→    location: str(p.location) || null,
   218→    status: str(p.status, existing?.status ?? 'active'),
   219→  }
   220→  if (!existing) {
   221→    if (!data.categoryId) return 'skipped'
   222→    await db.medicine.create({ data: { id, sku: str(p.sku) || `SYNC-${id.slice(-8)}`, ...data, stockQty: num(p.stockQty, 0) } })
   223→  } else {
   224→    // Never let a replicated op reduce local stock (stock is node-local).
   225→    await db.medicine.update({ where: { id }, data })
   226→  }
   227→  publishLive(['medicine', 'sync'], `Catalog update from ${opts.asHQ ? 'branch' : 'HQ'} applied`, 'sync')
   228→  return 'ok'
   229→}

   231→// --- SKU-keyed catalog + stock ops (copy-catalog / inventory-file import) ----
   232→// The emitting node may not share medicine ids with the receiver, so these ops
   233→// resolve rows by SKU → barcode → name inside the branch named in the payload
   234→// (auto-provisioned by code when unknown). Catalog travels, stock does not:
   235→// an upsert never touches quantities on an existing row — only stock.delta /
   236→// stock.set move stock, and both leave an inventoryLog keyed by the wire op id
   237→// so a re-delivered op is a clean duplicate, never a double-apply.
   238→
   239→async function branchIdForCode(p: Record<string, unknown>): Promise<string | null> {
   240→  const code = str(p.branchCode).trim()
   241→  if (!code) return null
   242→  const existing = await db.branch.findFirst({ where: { code }, select: { id: true } })
   243→  if (existing) return existing.id
   244→  const created = await db.branch.create({
   245→    data: { code, name: str(p.branchName).trim() || code, isActive: true },
   246→    select: { id: true },
   247→  })
   248→  return created.id
   249→}

   251→async function categoryFor(name: string): Promise<string> {
   252→  const trimmed = name.trim() || 'General'
   253→  const hit = await db.category.findUnique({ where: { name: trimmed }, select: { id: true } })
   254→  if (hit) return hit.id
   255→  const created = await db.category.create({ data: { name: trimmed }, select: { id: true } })
   256→  return created.id
   257→}

   259→async function resolveMedicineBySku(branchId: string | null, p: Record<string, unknown>) {
   260→  const sku = str(p.sku)
   261→  const barcode = str(p.barcode)
   262→  const name = str(p.name)
   263→  const scope = branchId ? { branchId } : {}
   264→  if (sku) {
   265→    const bySku = await db.medicine.findFirst({ where: { sku, ...scope } })
   266→    if (bySku) return bySku
   267→  }
   268→  if (barcode) {
   269→    const byBarcode = await db.medicine.findFirst({ where: { barcode, ...scope } })
   270→    if (byBarcode) return byBarcode
   271→  }
   272→  if (name) {
   273→    const byName = await db.medicine.findFirst({ where: { name, ...scope } })
   274→    if (byName) return byName
   275→  }
   276→  return null
   277→}

   279→async function applyCatalogUpsertBySku(
   280→  op: WireOp,
   281→  p: Record<string, unknown>,
   282→  opts: { asHQ: boolean },
   283→): Promise<ApplyResult> {
   284→  let branchId = await branchIdForCode(p)
   285→  let existing = await resolveMedicineBySku(branchId, p)
   286→  if (!existing && !branchId) {
   287→    // No branch context — fall back to the row the globally-unique SKU names.
   288→    existing = await db.medicine.findUnique({ where: { sku: str(p.sku) } })
   289→    branchId = existing?.branchId ?? null
   290→  }
   291→  if (!branchId) return 'skipped'
   292→  const data = {
   293→    name: str(p.name, existing?.name ?? 'Unnamed'),
   294→    genericName: str(p.genericName) || null,
   295→    brand: str(p.brand) || null,
   296→    unit: str(p.unit, existing?.unit ?? 'Box'),
   297→    barcode: str(p.barcode) || existing?.barcode || null,
   298→    costPrice: num(p.costPrice, existing?.costPrice ?? 0),
   299→    sellingPrice: num(p.sellingPrice, existing?.sellingPrice ?? 0),
   300→    prescriptionRequired: p.prescriptionRequired === true,
   301→    reorderLevel: Math.max(0, Math.floor(num(p.reorderLevel, existing?.reorderLevel ?? 20))),
   302→    location: str(p.location) || existing?.location || null,
   303→    // Formulary flag only travels when the emitter sent it — never false-reset
   304→    ...(p.insuranceCovered !== undefined ? { insuranceCovered: p.insuranceCovered === true } : {}),
   305→  }
   306→  const note = `Catalog update from ${opts.asHQ ? 'branch' : 'HQ'} applied`
   307→  if (existing) {
   308→    // Catalog travels, stock does not — never touch quantities here.
   309→    await db.medicine.update({ where: { id: existing.id }, data })
   310→    publishLive(['medicine', 'sync'], note, 'sync')
   311→    return 'ok'
   312→  }
   313→  const sku = str(p.sku) || `SYNC-${(op.targetKey || op.id).slice(-8)}`
   314→  const taken = await db.medicine.findUnique({ where: { sku }, select: { id: true } })
   315→  const branchCode = (await db.branch.findUnique({ where: { id: branchId }, select: { code: true } }))?.code || 'BR'
   316→  const finalSku = taken ? await deriveSkuForBranch(sku, branchCode) : sku
   317→  const categoryId = await categoryFor(str(p.category, 'General'))
   318→  await db.medicine.create({
   319→    data: {
   320→      ...data,
   321→      sku: finalSku,
   322→      categoryId,
   323→      branchId,
   324→      status: 'active',
   325→      stockQty: Math.max(0, Math.floor(num(p.initialQty, num(p.stockQty, 0)))),
   326→    },
   327→  })
   328→  publishLive(['medicine', 'sync'], note, 'sync')
   329→  return 'ok'
   330→}

   332→async function applyStockOpBySku(op: WireOp, p: Record<string, unknown>): Promise<ApplyResult> {
   333→  // Replay guard: the wire op id is stable across re-delivery.
   334→  const dup = await db.inventoryLog.findFirst({ where: { reference: op.id, type: 'import' } })
   335→  if (dup) return 'duplicate'
   336→  let branchId = await branchIdForCode(p)
   337→  let med = await resolveMedicineBySku(branchId, p)
   338→  if (!med && !branchId) med = await db.medicine.findUnique({ where: { sku: str(p.sku) } })
   339→  if (!med) return 'skipped'
   340→  const isSet = op.op === 'stock.set'
   341→  const nextQty = isSet
   342→    ? Math.max(0, Math.floor(num(p.qty, med.stockQty)))
   343→    : Math.max(0, med.stockQty + Math.floor(num(p.delta, 0)))
   344→  const applied = nextQty - med.stockQty
   345→  await db.$transaction([
   346→    db.medicine.update({ where: { id: med.id }, data: { stockQty: nextQty } }),
   347→    db.inventoryLog.create({
   348→      data: {
   349→        medicineId: med.id,
   350→        type: 'import',
   351→        quantity: applied,
   352→        reason: str(p.reason, isSet ? 'Synced stock set' : 'Synced stock adjustment'),
   353→        reference: op.id,
   354→      },
   355→    }),
   356→  ])
   357→  publishLive(['stock', 'sync'], `Stock ${isSet ? 'set' : 'adjustment'} applied from sync`, 'sync')
   358→  return 'ok'
   359→}

   363→async function applyInventoryAdjusted(
   364→  op: WireOp,
   365→  p: Record<string, unknown>,
   366→): Promise<ApplyResult> {
   367→  const originId = str(p.logId) || op.id
   368→  const dup = await db.inventoryLog.findUnique({ where: { id: originId } }).catch(() => null)
   369→  if (dup) return 'duplicate'
   370→  const medicineId = str(p.medicineId)
   371→  const delta = num(p.quantity)
   372→  if (!medicineId || delta === 0) return 'skipped'
   373→  const med = await db.medicine.findUnique({ where: { id: medicineId } })
   374→  if (!med) return 'skipped'
   375→  await db.$transaction([
   376→    db.medicine.update({ where: { id: medicineId }, data: { stockQty: { increment: delta } } }),
   377→    db.inventoryLog.create({
   378→      data: {
   379→        id: originId,
   380→        medicineId,
   381→        type: str(p.type, 'adjustment'),
   382→        quantity: delta,
   383→        reason: str(p.reason, 'Synced adjustment'),
   384→        reference: str(p.reference) || null,
   385→      },
   386→    }),
   387→  ])
   388→  publishLive(['stock', 'sync'], `Stock adjustment synced (${delta > 0 ? '+' : ''}${delta})`, 'sync')
   389→  return 'ok'
   390→}

   483→/** Mirrors a write-off / supplier return / quarantine from another node. */
   484→async function applyStockReturned(op: WireOp, p: Record<string, unknown>): Promise<ApplyResult> {
   485→  const items = Array.isArray(p.items) ? p.items : []
   486→  const returnNo = str(p.returnNo, op.targetKey || 'RET')
   487→  let applied = 0
   488→  for (const it of items) {
   489→    const item = it as Record<string, unknown>
   490→    const mid = str(item.medicineId)
   491→    const qty = Math.trunc(num(item.qty, 0))
   492→    if (!mid || qty <= 0) continue
   493→    const med = await db.medicine.findUnique({ where: { id: mid } })
   494→    if (!med) continue
   495→    const fromQuarantine = item.fromQuarantine === true
   496→    if (fromQuarantine) {
   497→      if (med.quarantineQty < qty) continue
   498→      const action = str(item.action, 'writeoff')
   499→      await db.medicine.update({
   500→        where: { id: mid },
   501→        data: { quarantineQty: { decrement: qty } },
   502→      })
   503→      if (action !== 'restore') {
   504→        await db.inventoryLog.create({ data: { medicineId: mid, type: 'quarantine_out', quantity: -qty, reason: `Quarantine ${action} (synced ${returnNo})`, reference: returnNo } })
   505→      }
   506→      applied++
   507→      continue
   508→    }
   509→    if (med.stockQty < qty) continue
   510→    if (item.action === 'quarantine') {
   511→      await db.medicine.update({ where: { id: mid }, data: { stockQty: { decrement: qty }, quarantineQty: { increment: qty } } })
   512→    } else {
   513→      await db.medicine.update({ where: { id: mid }, data: { stockQty: { decrement: qty } } })
   514→      await db.inventoryLog.create({ data: { medicineId: mid, type: str(item.type, 'writeoff'), quantity: -qty, reason: `${str(item.reason, 'writeoff')} (synced ${returnNo})`, reference: returnNo } })
   515→    }
   516→    applied++
   517→  }
   518→  if (applied > 0) publishLive(['stock', 'sync'], `Stock adjustment synced from peer (${returnNo})`, 'sync')
   519→  return applied > 0 ? 'ok' : 'skipped'
   520→}

   539→async function applyExpenseCreated(op: WireOp, p: Record<string, unknown>, opts: { asHQ: boolean }): Promise<ApplyResult> {
   540→  const originId = str(p.id) || op.id
   541→  const existing = await db.expense.findUnique({ where: { id: originId } }).catch(() => null)
   542→  if (existing) return 'duplicate'
   543→  const amount = num(p.amount)
   544→  if (amount <= 0) return 'skipped'
   545→
   546→  // Per-node document numbering can collide across nodes (each branch counts
   547→  // its own rows) — mirror the sale convention: prefix the origin node id.
   548→  const localNo = opts.asHQ && op.nodeId !== 'HQ'
   549→    ? `${op.nodeId}:${str(p.expenseNo)}`
   550→    : str(p.expenseNo)
   551→
   552→  // Categories are node-local: resolve by id, then name, then create it so the
   553→  // replicated expense always lands with its category intact.
   554→  let categoryId = str(p.categoryId)
   555→  const categoryName = str(p.categoryName, 'Other')
   556→  const catById = categoryId ? await db.expenseCategory.findUnique({ where: { id: categoryId } }).catch(() => null) : null
   557→  if (!catById) {
   558→    const byName = await db.expenseCategory.findUnique({ where: { name: categoryName } }).catch(() => null)
   559→    if (byName) {
   560→      categoryId = byName.id
   561→    } else {
   562→      const created = await db.expenseCategory.create({ data: { name: categoryName } })
   563→      categoryId = created.id
   564→    }
   565→  }
   566→
   567→  await db.expense.create({
   568→    data: {
   569→      id: originId,
   570→      expenseNo: localNo || `SYNC-${originId.slice(-8)}`,
   571→      categoryId,
   572→      branchId: str(p.branchId) || null,
   573→      description: str(p.description, 'Synced expense'),
   574→      amount,
   575→      paymentMethod: str(p.paymentMethod, 'cash'),
   576→      provider: str(p.provider) || null,
   577→      reference: str(p.reference) || null,
   578→      status: str(p.status, 'completed'),
   579→      note: str(p.note) || null,
   580→      recordedByName: str(p.recordedByName, 'Sync'),
   581→      clientRef: `sync:${originId}`,
   582→      createdAt: date(p.createdAt) ?? new Date(),
   583→    },
   584→  })
   585→  publishLive(['sync', 'expense'], `Expense ${localNo} synced from ${op.nodeId}`, 'sync')
   586→  return 'ok'
   587→}

   606→async function applyQuotationUpdated(op: WireOp, p: Record<string, unknown>, opts: { asHQ: boolean }): Promise<ApplyResult> {
   607→  const originId = str(p.id) || op.id
   608→  if (!originId) return 'skipped'
   609→  const items = Array.isArray(p.items) ? p.items : []
   610→
   611→  // Only link the customer when the receiving node actually has them.
   612→  const customerId = str(p.customerId)
   613→  const customer = customerId ? await db.customer.findUnique({ where: { id: customerId } }).catch(() => null) : null
   614→
   615→  const data = {
   616→    quoteNo: opts.asHQ && op.nodeId !== 'HQ'
   617→      ? `${op.nodeId}:${str(p.quoteNo)}`
   618→      : str(p.quoteNo, `QTN-${originId.slice(-8)}`),
   619→    customerId: customer?.id ?? null,
   620→    customerName: str(p.customerName) || customer?.name || null,
   621→    branchId: str(p.branchId) || null,
   622→    status: str(p.status, 'draft'),
   623→    subtotal: num(p.subtotal),
   624→    discount: num(p.discount),
   625→    tax: num(p.tax),
   626→    total: num(p.total),
   627→    validUntil: date(p.validUntil),
   628→    note: str(p.note) || null,
   629→    createdByName: str(p.createdByName, 'Sync'),
   630→  }
   631→
   632→  await db.$transaction(async (tx) => {
   633→    const existing = await tx.quotation.findUnique({ where: { id: originId } })
   634→    if (existing) {
   635→      // A status transition already applied here wins over a stale snapshot —
   636→      // never regress converted/expired/cancelled states.
   637→      const sticky = ['converted', 'expired', 'cancelled']
   638→      if (sticky.includes(existing.status) || existing.updatedAt > (date(p.updatedAt) ?? new Date(0))) {
   639→        return
   640→      }
   641→      await tx.quotation.update({ where: { id: originId }, data })
   642→      await tx.quotationItem.deleteMany({ where: { quotationId: originId } })
   643→    } else {
   644→      await tx.quotation.create({ data: { id: originId, ...data, clientRef: `sync:${originId}` } })
   645→    }
   646→    for (const it of items) {
   647→      const item = it as Record<string, unknown>
   648→      const mid = str(item.medicineId)
   649→      const med = mid ? await tx.medicine.findUnique({ where: { id: mid }, select: { id: true } }).catch(() => null) : null
   650→      await tx.quotationItem.create({
   651→        data: {
   652→          quotationId: originId,
   653→          medicineId: med?.id ?? null,
   654→          name: str(item.name, 'Item'),
   655→          sku: str(item.sku) || null,
   656→          qty: Math.trunc(num(item.qty, 1)),
   657→          unitPrice: num(item.unitPrice),
   658→          discount: num(item.discount),
   659→          total: num(item.total),
   660→        },
   661→      })
   662→    }
   663→  })
   664→  publishLive(['sync', 'quotation'], `Quotation ${data.quoteNo} synced from ${opts.asHQ ? 'branch' : 'HQ'}`, 'sync')
   665→  return 'ok'
   666→}

   668→async function applyQuotationStatus(op: WireOp, p: Record<string, unknown>): Promise<ApplyResult> {
   669→  const id = str(p.id)
   670→  if (!id) return 'skipped'
   671→  const quote = await db.quotation.findUnique({ where: { id } })
   672→  if (!quote) return 'skipped'

   674→  if (status === quote.status && !str(p.convertedSaleId)) return 'duplicate'
   675→  await db.quotation.update({
   676→    where: { id },
   677→    data: {
   678→      status,
   679→      convertedSaleId: str(p.convertedSaleId) || quote.convertedSaleId,
   680→      convertedInvoice: str(p.convertedInvoice) || quote.convertedInvoice,
   681→      updatedAt: date(p.at) ?? new Date(),
   682→    },
   683→  })
   684→  return 'ok'
