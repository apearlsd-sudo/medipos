    32→    32→/** Tier 2: write-offs, supplier returns, the quarantine desk and the pre-expiry disposal workflow. */

    34→    34→  const [meds, setMeds] = useState<MedOption[]>([])
    35→    35→  const [suppliers, setSuppliers] = useState<SupplierOption[]>([])
    36→    36→  const [rows, setRows] = useState<ReturnRow[]>([])
    37→    37→  const [quar, setQuar] = useState<QuarantineRow[]>([])
    38→    38→  const [loading, setLoading] = useState(true)
    39→    39→
    40→    40→  const [noteOpen, setNoteOpen] = useState(false)
    41→    41→  const [type, setType] = useState('writeoff')
    42→    42→  const [reason, setReason] = useState('expired')
    43→    43→  const [supplierId, setSupplierId] = useState('none')
    44→    44→  const [note, setNote] = useState('')
    45→    45→  const [lines, setLines] = useState<DraftLine[]>([emptyDraft()])
    46→    46→  const [saving, setSaving] = useState(false)
    47→    47→  const [lastNote, setLastNote] = useState<{ returnNo: string; totalValue: number; itemCount: number } | null>(null)
    48→    48→
    49→    49→  // --- Pre-expiry disposal desk state ---
    50→    50→  const [cands, setCands] = useState<PreExpiryCandidate[]>([])
    51→    51→  const [peDays, setPeDays] = useState('60')
    52→    52→  const [peLoading, setPeLoading] = useState(true)
    53→    53→  const [peSel, setPeSel] = useState<Record<string, string>>({}) // medicineId -> dispose qty (string)
    54→    54→  const [peOpen, setPeOpen] = useState(false)
    55→    55→  const [peAction, setPeAction] = useState<'writeoff' | 'return_supplier'>('writeoff')
    56→    56→  const [peSupplier, setPeSupplier] = useState('none')
    57→    57→  const [peNote, setPeNote] = useState('')
    58→    58→  const [peSaving, setPeSaving] = useState(false)
    59→    59→
    60→    60→  const load = useCallback(async () => {
    61→    61→    setLoading(true)
    62→    62→    try {
    63→    63→      const [mRes, sRes, rRes, qRes] = await Promise.all([
    64→    64→        fetch('/api/medicines?status=active', { cache: 'no-store' }),
    65→    65→        fetch('/api/suppliers', { cache: 'no-store' }),
    66→    66→        fetch('/api/stock-returns', { cache: 'no-store' }),
    67→    67→        fetch('/api/quarantine', { cache: 'no-store' }),
    68→    68→      ])
    69→    69→      setMeds((await mRes.json()).medicines ?? [])
    70→    70→      setSuppliers((await sRes.json()).suppliers ?? [])
    71→    71→      setRows((await rRes.json()).returns ?? [])
    72→    72→      setQuar((await qRes.json()).items ?? [])
    73→    73→    } finally {
    74→    74→      setLoading(false)
    75→    75→    }
    76→    76→  }, [])
    77→    77→
    78→    78→  const loadCandidates = useCallback(async (days: string) => {
    79→    79→    setPeLoading(true)
    80→    80→    try {
    81→    81→      const res = await fetch(`/api/disposals/pre-expiry?days=${days}`, { cache: 'no-store' })
    82→    82→      const json = await res.json()
    83→    83→      setCands(res.ok ? (json.items ?? []) : [])
    84→    84→    } catch {
    85→    85→      setCands([])
    86→    86→    } finally {
    87→    87→      setPeLoading(false)
    88→    88→    }
    89→    89→  }, [])
    90→    90→
    91→    91→  useEffect(() => { load() }, [load])
    92→    92→  useEffect(() => { setPeSel({}); loadCandidates(peDays) }, [loadCandidates, peDays])
    93→    93→
    94→    94→  const openNote = () => {
    95→    95→    setType('writeoff'); setReason('expired'); setSupplierId('none'); setNote('')
    96→    96→    setLines([emptyDraft()]); setLastNote(null); setNoteOpen(true)
    97→    97→  }
    98→    98→
    99→    99→  const addLine = () => setLines((ls) => [...ls, emptyDraft()])
   100→   100→  const updateLine = (i: number, patch: Partial<DraftLine>) => setLines((ls) => ls.map((l, j) => (j === i ? { ...l, ...patch } : l)))
   101→   101→  const removeLine = (i: number) => setLines((ls) => (ls.length > 1 ? ls.filter((_, j) => j !== i) : ls))
   102→   102→
   103→   103→  const draftValue = lines.reduce((s, l) => {
   104→   104→    const m = meds.find((x) => x.id === l.medicineId)
   105→   105→    return m ? s + (Number(l.qty) || 0) * m.costPrice : s
   106→   106→  }, 0)
   107→   107→
   108→   108→  const submitNote = async () => {
   109→   109→    const items = lines
   110→   110→      .filter((l) => l.medicineId && Number(l.qty) > 0)
   111→   111→      .map((l) => ({ medicineId: l.medicineId, qty: Number(l.qty), action: type === 'supplier_return' ? 'return_supplier' : 'writeoff', fromQuarantine: l.fromQuarantine }))
   112→   112→    if (items.length === 0) { toast({ title: 'Add at least one item', variant: 'destructive' }); return }
   113→   113→    setSaving(true)
   114→   114→    try {
   115→   115→      const res = await fetch('/api/stock-returns', {
   116→   116→        method: 'POST',
   117→   117→        headers: { 'Content-Type': 'application/json' },
   118→   118→        body: JSON.stringify({ type, reason, supplierId: type === 'supplier_return' ? supplierId : undefined, note: note || undefined, items }),
   119→   119→      })
   120→   120→      const json = await res.json()
   121→   121→      if (!res.ok) { toast({ title: 'Could not create the note', description: json.error, variant: 'destructive' }); return }
   122→   122→      setLastNote({ returnNo: json.return.returnNo, totalValue: json.return.totalValue, itemCount: json.return.items.length })
   123→   123→      toast({ title: `${json.return.returnNo} recorded`, description: `${json.return.items.length} line(s) · ${fmtMoney(json.return.totalValue)} cost value removed.` })
   124→   124→      load()
   125→   125→    } finally {
   126→   126→      setSaving(false)
   127→   127→    }
   128→   128→  }
   129→   129→
   130→   130→  const cancelNote = async (r: ReturnRow) => {
   131→   131→    if (!confirm(`Cancel ${r.returnNo} and put the stock back?`)) return
   132→   132→    const res = await fetch(`/api/stock-returns/${r.id}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: '{}' })
   133→   133→    const json = await res.json()
   134→   134→    if (!res.ok) { toast({ title: 'Cancel failed', description: json.error, variant: 'destructive' }); return }
   135→   135→    toast({ title: `${r.returnNo} cancelled — stock restored` })
   136→   136→    load()
   137→   137→  }
   138→   138→
   139→   139→  const resolveQuarantine = async (q: QuarantineRow, action: 'restore' | 'writeoff' | 'return_supplier', qty: number) => {
   140→   140→    const res = await fetch('/api/quarantine', {
   141→   141→      method: 'POST',
   142→   142→      headers: { 'Content-Type': 'application/json' },
   143→   143→      body: JSON.stringify({ medicineId: q.id, action, qty }),
   144→   144→    })
   145→   145→    const json = await res.json()
   146→   146→    if (!res.ok) { toast({ title: 'Action failed', description: json.error, variant: 'destructive' }); return }
   147→   147→    toast({ title: action === 'restore' ? 'Back on the shelf' : action === 'writeoff' ? 'Written off' : 'Marked for supplier return', description: `${q.name} × ${qty}` })
   148→   148→    load()
   149→   149→  }
   150→   150→
   151→   151→  // --- Pre-expiry disposal desk helpers ---
   152→   152→  const selectedIds = Object.keys(peSel).filter((id) => Number(peSel[id]) > 0 && cands.some((c) => c.id === id))
   153→   153→  const selectedRows = selectedIds.map((id) => cands.find((c) => c.id === id)!).filter(Boolean)
   154→   154→  const selUnits = selectedRows.reduce((s, c) => s + (Number(peSel[c.id]) || 0), 0)
   155→   155→  const selValue = selectedRows.reduce((s, c) => s + (Number(peSel[c.id]) || 0) * c.costPrice, 0)
   156→   156→  const allSelected = cands.length > 0 && cands.every((c) => Number(peSel[c.id]) > 0)
   157→   157→  const selSuppliers = [...new Set(selectedRows.map((c) => c.supplierId).filter(Boolean))]
   158→   158→  const candUnits = cands.reduce((s, c) => s + c.stockQty, 0)
   159→   159→  const candValue = cands.reduce((s, c) => s + c.valueAtRisk, 0)
   160→   160→
   161→   161→  const toggleRow = (c: PreExpiryCandidate, on: boolean) => {
   162→   162→    setPeSel((prev) => {
   163→   163→      const next = { ...prev }
   164→   164→      if (on) next[c.id] = String(c.stockQty)
   165→   165→      else delete next[c.id]
   166→   166→      return next
   167→   167→    })
   168→   168→  }
   169→   169→  const toggleAll = () => {
   170→   170→    if (allSelected) { setPeSel({}); return }
   171→   171→    const next: Record<string, string> = {}
   172→   172→    for (const c of cands) next[c.id] = String(c.stockQty)
   173→   173→    setPeSel(next)
   174→   174→  }
   175→   175→
   176→   176→  const openDisposal = (action: 'writeoff' | 'return_supplier') => {
   177→   177→    if (selectedIds.length === 0) { toast({ title: 'Select at least one product first', variant: 'destructive' }); return }
   178→   178→    setPeAction(action)
   179→   179→    setPeNote('')
   180→   180→    // Preselect the supplier when the whole selection comes from one source.
   181→   181→    setPeSupplier(action === 'return_supplier' && selSuppliers.length === 1 ? (selSuppliers[0] as string) : 'none')
   182→   182→    setPeOpen(true)
   183→   183→  }
   184→   184→
   185→   185→  const submitDisposal = async () => {
   186→   186→    const items = selectedIds.map((id) => ({ medicineId: id, qty: Number(peSel[id]) }))
   187→   187→    if (items.length === 0) { toast({ title: 'Select at least one product', variant: 'destructive' }); return }
   188→   188→    if (peAction === 'return_supplier' && peSupplier === 'none') { toast({ title: 'Choose a supplier for the return', variant: 'destructive' }); return }
   189→   189→    setPeSaving(true)
   190→   190→    try {
   191→   191→      const res = await fetch('/api/disposals/pre-expiry', {
   192→   192→        method: 'POST',
   193→   193→        headers: { 'Content-Type': 'application/json' },
   194→   194→        body: JSON.stringify({
   195→   195→          action: peAction,
   196→   196→          days: Number(peDays),
   197→   197→          items,
   198→   198→          supplierId: peAction === 'return_supplier' ? peSupplier : undefined,
   199→   199→          note: peNote || undefined,
   200→   200→        }),
   201→   201→      })
   202→   202→      const json = await res.json()
   203→   203→      if (!res.ok) { toast({ title: 'Disposal failed', description: json.error, variant: 'destructive' }); return }
   204→   204→      const destroyed = peAction === 'writeoff'
   205→   205→      toast({
   206→   206→        title: `${json.return.returnNo} recorded — disposed ${json.return.items.reduce((s: number, i: { qty: number }) => s + i.qty, 0)} unit(s) before expiry`,
   207→   207→        description: destroyed
   208→   208→          ? `${fmtMoney(json.return.totalValue)} written off and auto-booked as an Inventory Write-offs expense${json.expenseNo ? ` (${json.expenseNo})` : ''}.`
   209→   209→          : `${fmtMoney(json.return.totalValue)} left with the supplier return note — recoverable via credit, no P&L hit.`,
   210→   210→      })
   211→   211→      setPeOpen(false)
   212→   212→      setPeSel({})
   213→   213→      loadCandidates(peDays)
   214→   214→      load()
   215→   215→    } finally {
   216→   216→      setPeSaving(false)
   217→   217→    }
   218→   218→  }
   219→   219→
   220→   220→  const daysLeftChip = (d: number) => (
   221→   221→    <span className={`inline-flex rounded-full px-2 py-0.5 text-xs font-semibold tabular-nums ${
   222→   222→      d <= 14
   223→   223→        ? 'bg-red-50 text-red-700 dark:bg-red-500/10 dark:text-red-400'
   224→   224→        : d <= 30
   225→   225→          ? 'bg-amber-50 text-amber-700 dark:bg-amber-500/10 dark:text-amber-400'
   226→   226→          : 'bg-sky-50 text-sky-700 dark:bg-sky-500/10 dark:text-sky-400'
   227→   227→    }`}>
   228→   228→      {d} d
   229→   229→    </span>
   230→   230→  )
   231→   231→
   232→   232→  return (
   233→   233→    <Tabs defaultValue="notes">
   234→   234→      <TabsList className="rounded-full">
   235→   235→        <TabsTrigger value="notes" className="rounded-full">Write-offs &amp; Returns</TabsTrigger>
   236→   236→        <TabsTrigger value="pre" className="rounded-full">Pre-Expiry Review ({cands.length})</TabsTrigger>
   237→   237→        <TabsTrigger value="quarantine" className="rounded-full">Quarantine ({quar.length})</TabsTrigger>
   238→   238→      </TabsList>
   239→   239→
   240→   240→      <TabsContent value="notes" className="mt-3 space-y-3">
   241→   241→        <div className="flex justify-end">
   242→   242→          <Button onClick={openNote} className="rounded-full bg-teal-600 hover:bg-teal-700"><Plus className="h-4 w-4" /> New Write-off / Return Note</Button>
   243→   243→        </div>
   244→   244→        {lastNote && (
   245→   245→          <Card className="border-teal-300 bg-teal-50/60 p-4 text-sm dark:border-teal-500/30 dark:bg-teal-500/10">
   246→   246→            <p className="font-semibold">{lastNote.returnNo} recorded — {lastNote.itemCount} line(s), {fmtMoney(lastNote.totalValue)} cost value removed from stock.</p>
   247→   247→            <p className="text-xs text-muted-foreground">The movement is in the audit log{type === 'supplier_return' ? '; print or send the note to the supplier for credit.' : '.'}</p>
   248→   248→          </Card>
   249→   249→        )}
   250→   250→        <Card>
   251→   251→          <Table>
   252→   252→            <TableHeader>
   253→   253→              <TableRow className="hover:bg-transparent">
   254→   254→                <TableHead>Note</TableHead><TableHead>Type</TableHead><TableHead>Reason</TableHead>
   255→   255→                <TableHead className="text-center">Units</TableHead><TableHead className="text-right">Cost Value</TableHead>
   256→   256→                <TableHead className="hidden md:table-cell">By</TableHead><TableHead className="hidden md:table-cell">When</TableHead><TableHead />
   257→   257→              </TableRow>
   258→   258→            </TableHeader>
   259→   259→            <TableBody>
   260→   260→              {rows.map((r) => (
   261→   261→                <TableRow key={r.id}>
   262→   262→                  <TableCell className="font-medium">{r.returnNo}</TableCell>
   263→   263→                  <TableCell><Badge variant="outline" className="rounded-full">{r.type === 'supplier_return' ? 'supplier return' : 'write-off'}</Badge></TableCell>
   264→   264→                  <TableCell className="capitalize">
   265→   265→                    {r.reason.replace(/_/g, '-')}{r.source === 'pre_expiry' && (
   266→   266→                      <Badge className="ml-1.5 rounded-full bg-teal-100 text-[10px] uppercase tracking-wide text-teal-700 dark:bg-teal-500/15 dark:text-teal-300">pre-expiry desk</Badge>
   267→   267→                    )}
   268→   268→                    {r.supplier ? <span className="block text-xs text-muted-foreground">{r.supplier}</span> : null}
   269→   269→                  </TableCell>
   270→   270→                  <TableCell className="text-center tabular-nums">{r.units}</TableCell>
   271→   271→                  <TableCell className="text-right tabular-nums font-semibold">{fmtMoney(r.totalValue)}</TableCell>
   272→   272→                  <TableCell className="hidden md:table-cell text-sm text-muted-foreground">{r.createdByName}</TableCell>
   273→   273→                  <TableCell className="hidden md:table-cell text-xs text-muted-foreground">{fmtRelative(r.createdAt)}</TableCell>
   274→   274→                  <TableCell>
   275→   275→                    {r.status === 'completed' ? (
   276→   276→                      <Button variant="ghost" size="icon" className="h-7 w-7 text-red-500" onClick={() => cancelNote(r)} aria-label="Cancel note"><RotateCcw className="h-3.5 w-3.5" /></Button>
   277→   277→                    ) : <Badge variant="outline" className="rounded-full">cancelled</Badge>}
   278→   278→                  </TableCell>
   279→   279→                </TableRow>
   280→   280→              ))}
   281→   281→              {!loading && rows.length === 0 && <TableRow><TableCell colSpan={8} className="py-10 text-center text-muted-foreground">No write-offs or returns yet — expired or damaged stock recorded here keeps the audit trail clean.</TableCell></TableRow>}
   282→   282→            </TableBody>
   283→   283→          </Table>
   284→   284→        </Card>
   285→   285→      </TabsContent>
   286→   286→
   287→   287→      <TabsContent value="pre" className="mt-3 space-y-3">
   288→   288→        <Card>
   289→   289→          <div className="flex flex-wrap items-center gap-2 border-b p-3 text-sm font-medium">
   290→   290→            <CalendarClock className="h-4 w-4 text-teal-600" />
   291→   291→            <span>Pre-expiry disposal desk — catch stock <em>before</em> the expiry date: return it to the supplier for credit, or write it off early</span>
   292→   292→            <div className="ml-auto flex items-center gap-2">
   293→   293→              <Select value={peDays} onValueChange={setPeDays}>
   294→   294→                <SelectTrigger className="h-8 w-[170px] rounded-full text-xs" aria-label="Expiry window"><SelectValue /></SelectTrigger>
   295→   295→                <SelectContent>
   296→   296→                  {PE_WINDOWS.map((w) => <SelectItem key={w} value={w}>Expiring within {w} days</SelectItem>)}
   297→   297→                </SelectContent>
   298→   298→              </Select>
   299→   299→              <Button variant="outline" size="icon" className="h-8 w-8 rounded-full" onClick={() => loadCandidates(peDays)} aria-label="Refresh candidates"><RotateCcw className="h-3.5 w-3.5" /></Button>
   300→   300→            </div>
   301→   301→          </div>
   302→   302→          {!peLoading && (
   303→   303→            <div className="flex flex-wrap items-center gap-x-4 gap-y-1 border-b bg-muted/40 px-4 py-2 text-xs text-muted-foreground">
   304→   304→              <span><strong className="text-foreground tabular-nums">{cands.length}</strong> product(s)</span>
   305→   305→              <span><strong className="text-foreground tabular-nums">{candUnits}</strong> {cands[0]?.unit ?? 'unit'}(s) on hand</span>
   306→   306→              <span><strong className="text-foreground tabular-nums">{fmtMoney(candValue)}</strong> cost value at risk</span>
   307→   307→              {selectedIds.length > 0 && (
   308→   308→                <span className="rounded-full bg-teal-600 px-2.5 py-0.5 font-semibold text-white tabular-nums">
   309→   309→                  {selectedIds.length} selected · {selUnits} unit(s) · {fmtMoney(selValue)}
   310→   310→                </span>
   311→   311→              )}
   312→   312→            </div>
   313→   313→          )}
   314→   314→          {peLoading ? (
   315→   315→            <p className="py-10 text-center text-sm text-muted-foreground">Scanning the shelves…</p>
   316→   316→          ) : cands.length === 0 ? (
   317→   317→            <p className="py-10 text-center text-sm text-muted-foreground">Nothing expiring within {peDays} days — the desk is clear. Raise the window to look further ahead.</p>
   318→   318→          ) : (
   319→   319→            <Table>
   320→   320→              <TableHeader>
   321→   321→                <TableRow className="hover:bg-transparent">
   322→   322→                  <TableHead className="w-10">
   323→   323→                    <Checkbox checked={allSelected} onCheckedChange={() => toggleAll()} aria-label="Select all candidates" />
   324→   324→                  </TableHead>
   325→   325→                  <TableHead>Product</TableHead>
   326→   326→                  <TableHead className="hidden md:table-cell">Batch</TableHead>
   327→   327→                  <TableHead>Expiry</TableHead>
   328→   328→                  <TableHead className="text-center">On hand</TableHead>
   329→   329→                  <TableHead className="text-right hidden md:table-cell">Unit cost</TableHead>
   330→   330→                  <TableHead className="text-right">Value at risk</TableHead>
   331→   331→                  <TableHead className="hidden md:table-cell">Dispose qty</TableHead>
   332→   332→                  <TableHead className="hidden lg:table-cell">Supplier</TableHead>
   333→   333→                </TableRow>
   334→   334→              </TableHeader>
