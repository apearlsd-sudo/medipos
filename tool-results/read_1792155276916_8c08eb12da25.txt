     1→'use client'
     2→
     3→import { Button } from '@/components/ui/button'
     4→import { Card } from '@/components/ui/card'
     5→import { Input } from '@/components/ui/input'
     6→import { Progress } from '@/components/ui/progress'
     7→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
     8→import { Skeleton } from '@/components/ui/skeleton'
     9→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    10→import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs'
    11→import { PageHeader, StatusBadge } from '@/components/pos/shared'
    12→import { VarianceBadge, VarianceValueBadge, signedQty } from '@/components/pos/stock-take-badges'
    13→import {
    14→  CancelTakeDialog, CompleteTakeDialog, CountSheetDialog, PauseTakeDialog, StartTakeDialog, TakeDetailDialog,
    15→  type StockTakeRec,
    16→} from '@/components/pos/stock-take-dialogs'
    17→import { fmtCompact, fmtDate, fmtDateTime, fmtMoney } from '@/lib/format'
    18→import type { StoreSettingsMap } from '@/lib/store-settings'
    19→import {
    20→  BarChart3, CalendarRange, ClipboardCheck, Download, Eye, FileText, History, ListChecks, PackageSearch,
    21→  PauseCircle, PlayCircle, RefreshCcw, Scale, TrendingDown, TrendingUp, XCircle,
    22→} from 'lucide-react'
    23→import { Bar, BarChart, CartesianGrid, Cell, ResponsiveContainer, Tooltip, XAxis, YAxis } from 'recharts'
    24→import { useCallback, useEffect, useMemo, useState } from 'react'
    25→import jsPDF from 'jspdf'
    26→import autoTable from 'jspdf-autotable'
    27→import { toast } from '@/hooks/use-toast'
    28→
    29→// ---------------------------------------------------------------------------
    30→// Module
    31→// ---------------------------------------------------------------------------
    32→
    33→export function StockTake({ storeSettings }: { storeSettings?: StoreSettingsMap | null }) {
    34→  const [takes, setTakes] = useState<StockTakeRec[]>([])
    35→  const [loading, setLoading] = useState(true)
    36→  const [tab, setTab] = useState<'history' | 'report'>('history')
    37→
    38→  // filters
    39→  const [search, setSearch] = useState('')
    40→  const [statusFilter, setStatusFilter] = useState('all')
    41→
    42→  // report
    43→  const todayIso = new Date().toISOString().slice(0, 10)
    44→  const monthAgoIso = new Date(Date.now() - 29 * 86400000).toISOString().slice(0, 10)
    45→  const [from, setFrom] = useState(monthAgoIso)
    46→  const [to, setTo] = useState(todayIso)
    47→  const [report, setReport] = useState<StockTakeReportData | null>(null)
    48→  const [reportLoading, setReportLoading] = useState(false)
    49→
    50→  // dialogs
    51→  const [startOpen, setStartOpen] = useState(false)
    52→  const [sheetOpen, setSheetOpen] = useState(false)
    53→  const [sheetTakeId, setSheetTakeId] = useState<string | null>(null)
    54→  const [completeOpen, setCompleteOpen] = useState(false)
    55→  const [completeTake, setCompleteTake] = useState<StockTakeRec | null>(null)
    56→  const [cancelOpen, setCancelOpen] = useState(false)
    57→  const [cancelTake, setCancelTake] = useState<StockTakeRec | null>(null)
    58→  const [pauseOpen, setPauseOpen] = useState(false)
    59→  const [pauseTake, setPauseTake] = useState<StockTakeRec | null>(null)
    60→  const [detailOpen, setDetailOpen] = useState(false)
    61→  const [detailId, setDetailId] = useState<string | null>(null)
    62→
    63→  const load = useCallback(async () => {
    64→    try {
    65→      const res = await fetch('/api/stock-takes', { cache: 'no-store' })
    66→      const json = await res.json()
    67→      setTakes(json.takes ?? [])
    68→    } finally {
    69→      setLoading(false)
    70→    }
    71→  }, [])
    72→
    73→  useEffect(() => { load() }, [load])
    74→
    75→  const loadReport = useCallback(async (f = from, t = to) => {
    76→    setReportLoading(true)
    77→    try {
    78→      const res = await fetch(`/api/stock-takes/report?from=${f}&to=${t}`, { cache: 'no-store' })
    79→      if (res.ok) setReport(await res.json())
    80→    } finally {
    81→      setReportLoading(false)
    82→    }
    83→  }, [from, to])
    84→
    85→  useEffect(() => {
    86→    if (tab === 'report') loadReport()
    87→  }, [tab, loadReport])
    88→
    89→  const active = useMemo(() => takes.find((t) => t.status === 'in_progress') ?? null, [takes])
    90→  const paused = useMemo(() => takes.find((t) => t.status === 'paused') ?? null, [takes])
    91→  const open = active ?? paused
    92→  const isPaused = !!paused
    93→
    94→  const filteredHistory = useMemo(() => {
    95→    let list = takes
    96→    if (statusFilter !== 'all') list = list.filter((t) => t.status === statusFilter)
    97→    if (search.trim()) {
    98→      const q = search.trim().toLowerCase()
    99→      list = list.filter((t) =>
   100→        t.stockTakeNo.toLowerCase().includes(q) ||
   101→        t.title.toLowerCase().includes(q) ||
   102→        t.startedByName.toLowerCase().includes(q))
   103→    }
   104→    return list
   105→  }, [takes, statusFilter, search])
   106→
   107→  const historyStats = useMemo(() => {
   108→    const completed = takes.filter((t) => t.status === 'completed')
   109→    const netValue = completed.reduce((n, t) => n + t.varianceValue, 0)
   110→    const lines = completed.reduce((n, t) => n + t.countedItems, 0)
   111→    const matched = completed.reduce((n, t) => n + t.matchedLines, 0)
   112→    return {
   113→      completed: completed.length,
   114→      cancelled: takes.filter((t) => t.status === 'cancelled').length,
   115→      paused: takes.filter((t) => t.status === 'paused').length,
   116→      netValue,
   117→      accuracy: lines > 0 ? Math.round((matched / lines) * 1000) / 10 : 0,
   118→    }
   119→  }, [takes])
   120→
   121→  const openCountSheet = (t: StockTakeRec) => { setSheetTakeId(t.id); setSheetOpen(true) }
   122→  const openComplete = (t: StockTakeRec) => { setCompleteTake(t); setCompleteOpen(true) }
   123→  const openCancel = (t: StockTakeRec) => { setCancelTake(t); setCancelOpen(true) }
   124→  const openPause = (t: StockTakeRec) => { setPauseTake(t); setPauseOpen(true) }
   125→  const openDetail = (id: string) => { setDetailId(id); setDetailOpen(true) }
   126→
   127→  const resumeTake = async (t: StockTakeRec) => {
   128→    try {
   129→      const res = await fetch(`/api/stock-takes/${t.id}/resume`, { method: 'PUT' })
   130→      const json = await res.json()
   131→      if (!res.ok) {
   132→        toast({ title: 'Cannot resume count', description: json.error ?? 'Unknown error', variant: 'destructive' })
   133→        return
   134→      }
   135→      toast({
   136→        title: `Count ${json.take.stockTakeNo} resumed`,
   137→        description: `${json.rebaselined ?? 0} uncounted line(s) re-baselined to current stock${json.added ? ` · ${json.added} new product(s) added to the sheet` : ''} — counting continues where you left off.`,
   138→      })
   139→      load()
   140→    } catch {
   141→      toast({ title: 'Network error', description: 'Session not resumed — try again.', variant: 'destructive' })
   142→    }
   143→  }
   144→
   145→  const applyPreset = (days: number) => {
   146→    const f = new Date(Date.now() - (days - 1) * 86400000).toISOString().slice(0, 10)
   147→    const t = new Date().toISOString().slice(0, 10)
   148→    setFrom(f)
   149→    setTo(t)
   150→    loadReport(f, t)
   151→  }
   152→
   153→  const exportCsv = () => {
   154→    if (!report) return
   155→    const rows: (string | number)[][] = [
   156→      ['Report', 'Stock Take Variance Report'],
   157→      ['Period', `${report.from} to ${report.to}`],
   158→      ['Sessions Completed', report.summary.takes],
   159→      ['Lines Counted', report.summary.linesCounted],
   160→      ['Count Accuracy %', report.summary.accuracyPct],
   161→      ['Net Units Variance', signedQty(report.summary.varianceQty)],
   162→      ['Net Value Variance', report.summary.varianceValue],
   163→      ['Shortage Value', report.summary.shortageValue],
   164→      ['Surplus Value', report.summary.surplusValue],
   165→      [],
   166→      ['SESSIONS'],
   167→      ['Take No', 'Title', 'Scope', 'Started By', 'Started', 'Completed', 'Lines', 'System Units', 'Counted Units', 'Net Qty Variance', 'Net Value Variance', 'Applied To Stock'],
   168→      ...report.takes.map((t) => [
   169→        t.stockTakeNo, t.title, t.scope === 'category' ? t.scopeCategory ?? 'category' : 'all',
   170→        t.startedByName, new Date(t.startedAt).toISOString(), t.completedAt ? new Date(t.completedAt).toISOString() : '',
   171→        t.totalItems, t.systemQty, t.countedQty, signedQty(t.varianceQty), t.varianceValue, t.applyToStock ? 'yes' : 'no',
   172→      ]),
   173→      [],
   174→      ['PRODUCT VARIANCES'],
   175→      ['Product', 'SKU', 'Takes', 'System Units', 'Counted Units', 'Net Qty Variance', 'Value Impact'],
   176→      ...report.topVariances.map((p) => [p.name, p.sku, p.takes, p.systemTotal, p.countedTotal, signedQty(p.varianceQty), p.varianceValue]),
   177→    ]
   178→    const csv = rows.map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(',')).join('\n')
   179→    const url = URL.createObjectURL(new Blob([csv], { type: 'text/csv;charset=utf-8' }))
   180→    const a = document.createElement('a')
   181→    a.href = url
   182→    a.download = `stock-take-report-${report.from}-to-${report.to}.csv`
   183→    a.click()
   184→    URL.revokeObjectURL(url)
   185→    toast({ title: 'Exported', description: 'Variance report written to CSV.' })
   186→  }
   187→
   188→  const exportPDF = () => {
   189→    if (!report) return
   190→    const storeName = storeSettings?.storeName || 'MediPOS Pharmacy'
   191→    const currency = storeSettings?.currencyCode || 'USD'
   192→
   193→    const doc = new jsPDF({ orientation: 'landscape', unit: 'mm', format: 'a4' })
   194→    const pageW = doc.internal.pageSize.getWidth()
   195→
   196→    // Header band
   197→    doc.setFillColor(13, 148, 136)
   198→    doc.rect(0, 0, pageW, 26, 'F')
   199→    doc.setTextColor(255, 255, 255)
   200→    doc.setFontSize(15)
   201→    doc.text(storeName, 14, 11)
   202→    doc.setFontSize(10)
   203→    doc.text(`Stock Take Variance Report — ${report.from} to ${report.to} (${currency})`, 14, 19)
   204→    doc.setFontSize(8)
   205→    doc.text(`Generated ${new Date().toLocaleString()}`, pageW - 14, 11, { align: 'right' })
   206→    doc.text('MediPOS Pro', pageW - 14, 19, { align: 'right' })
   207→
   208→    // Summary block
   209→    doc.setTextColor(40)
   210→    doc.setFontSize(10)
   211→    const s = report.summary
   212→    const left = [
   213→      `Sessions completed: ${s.takes}`,
   214→      `Lines counted: ${s.linesCounted}`,
   215→      `Count accuracy: ${s.accuracyPct}% matched`,
   216→    ]
   217→    const right = [
   218→      `Net units variance: ${signedQty(s.varianceQty)}`,
   219→      `Net value variance: ${fmtMoney(s.varianceValue)}`,
   220→      `Shortage ${fmtMoney(Math.abs(s.shortageValue))} · Surplus ${fmtMoney(s.surplusValue)}`,
   221→    ]
   222→    left.forEach((line, i) => doc.text(line, 14, 36 + i * 6))
   223→    right.forEach((line, i) => doc.text(line, pageW / 2 + 10, 36 + i * 6))
   224→
   225→    autoTable(doc, {
   226→      head: [['Take No', 'Title', 'Scope', 'Started By', 'Completed', 'Lines', 'System', 'Counted', 'Net Qty', 'Net Value', 'Applied']],
   227→      body: report.takes.map((t) => [
   228→        t.stockTakeNo, t.title, t.scope === 'category' ? t.scopeCategory ?? 'category' : 'All products',
   229→        t.startedByName, t.completedAt ? fmtDateTime(t.completedAt) : '—',
   230→        t.totalItems, t.systemQty, t.countedQty, signedQty(t.varianceQty), fmtMoney(t.varianceValue), t.applyToStock ? 'Yes' : 'No',
   231→      ]),
   232→      startY: 58,
   233→      styles: { fontSize: 8, cellPadding: 2 },
   234→      headStyles: { fillColor: [13, 148, 136], textColor: 255, fontStyle: 'bold' },
   235→      alternateRowStyles: { fillColor: [245, 250, 250] },
   236→      margin: { left: 14, right: 14 },
   237→    })
   238→
   239→    const afterFirst = (doc as unknown as { lastAutoTable?: { finalY: number } }).lastAutoTable?.finalY ?? 80
   240→    autoTable(doc, {
   241→      head: [['Product', 'SKU', 'Takes', 'System Units', 'Counted Units', 'Net Qty Variance', 'Value Impact']],
   242→      body: report.topVariances.map((p) => [
   243→        p.name, p.sku, p.takes, p.systemTotal, p.countedTotal, signedQty(p.varianceQty), fmtMoney(p.varianceValue),
   244→      ]),
   245→      startY: afterFirst + 8,
   246→      styles: { fontSize: 8, cellPadding: 2 },
   247→      headStyles: { fillColor: [13, 148, 136], textColor: 255, fontStyle: 'bold' },
   248→      alternateRowStyles: { fillColor: [245, 250, 250] },
   249→      margin: { left: 14, right: 14 },
   250→    })
   251→
   252→    const pages = doc.getNumberOfPages()
   253→    for (let i = 1; i <= pages; i++) {
   254→      doc.setPage(i)
   255→      doc.setFontSize(8)
   256→      doc.setTextColor(120)
   257→      doc.text(`Page ${i} of ${pages}`, pageW / 2, doc.internal.pageSize.getHeight() - 6, { align: 'center' })
   258→    }
   259→
   260→    doc.save(`stock-take-report-${report.from}-to-${report.to}.pdf`)
   261→  }
   262→
   263→  return (
   264→    <div className="space-y-6">
   265→      <PageHeader
   266→        title="Stock Take"
   267→        breadcrumb="Stock Take"
   268→        actions={
   269→          <>
   270→            <Button variant="outline" size="icon" className="rounded-full" onClick={load} aria-label="Refresh stock takes">
   271→              <RefreshCcw className="h-4 w-4" />
   272→            </Button>
   273→            <Button className="rounded-full bg-teal-600 hover:bg-teal-700" disabled={!!open} onClick={() => setStartOpen(true)}>
   274→              <PlayCircle className="h-4 w-4" /> Start Stock Take
   275→            </Button>
   276→          </>
   277→        }
   278→      />
   279→
   280→      {/* ============ Active count session ============ */}
   281→      {loading ? (
   282→        <Skeleton className="h-56 rounded-2xl" />
   283→      ) : open ? (
   284→        <Card className={`overflow-hidden ${isPaused ? 'border-amber-500/30' : 'border-teal-600/20'}`}>
   285→          <div className={`flex flex-col gap-3 border-b px-5 py-4 sm:flex-row sm:items-center sm:justify-between ${isPaused ? 'bg-gradient-to-r from-amber-500/10 via-amber-400/5 to-transparent' : 'bg-gradient-to-r from-teal-600/10 via-teal-500/5 to-transparent'}`}>
   286→            <div className="flex items-center gap-3">
   287→              <span className={`flex h-11 w-11 items-center justify-center rounded-xl text-white shadow-md ${isPaused ? 'bg-amber-500 shadow-amber-500/25' : 'bg-teal-600 shadow-teal-600/25'}`}>
   288→                {isPaused ? <PauseCircle className="h-5.5 w-5.5" /> : <ClipboardCheck className="h-5.5 w-5.5" />}
   289→              </span>
   290→              <div>
   291→                <div className="flex flex-wrap items-center gap-2">
   292→                  <h2 className="text-lg font-bold">{open.stockTakeNo}</h2>
   293→                  {isPaused ? <StatusBadge status="paused" /> : <StatusBadge status="active" />}
   294→                  {(open.resumeCount ?? 0) > 0 && (
   295→                    <span className="rounded-full bg-muted px-2 py-0.5 text-[10px] font-semibold">
   296→                      resumed {open.resumeCount}×{open.resumedAt ? ` · last ${fmtDateTime(open.resumedAt)}` : ''}
   297→                    </span>
   298→                  )}
   299→                  <span className="text-xs text-muted-foreground">
   300→                    {open.title} · {open.scope === 'category' ? open.scopeCategory : 'All products'} · by {open.startedByName} · {fmtDateTime(open.startedAt)}
   301→                  </span>
   302→                </div>
   303→                {isPaused ? (
   304→                  <p className="text-sm text-muted-foreground">
   305→                    Session paused{open.pausedAt ? ` ${fmtDateTime(open.pausedAt)}` : ''}{open.pausedByName ? ` by ${open.pausedByName}` : ''}{open.pauseNote ? ` — ${open.pauseNote}` : ''}.
   306→                    The shop keeps trading normally — resume any later day: uncounted lines re-baseline to current stock and products registered in the meantime join the sheet.
   307→                  </p>
   308→                ) : (
   309→                  <p className="text-sm text-muted-foreground">
   310→                    Physical count in progress — system stock was snapshotted when the session opened.
   311→                  </p>
   312→                )}
   313→              </div>
   314→            </div>
   315→            <div className="flex flex-wrap items-center gap-2">
   316→              {isPaused ? (
   317→                <Button size="sm" className="rounded-full bg-teal-600 hover:bg-teal-700" onClick={() => resumeTake(open)}>
   318→                  <PlayCircle className="h-4 w-4" /> Resume Counting
   319→                </Button>
   320→              ) : (
   321→                <Button size="sm" className="rounded-full bg-teal-600 hover:bg-teal-700" onClick={() => openCountSheet(open)}>
   322→                  <PackageSearch className="h-4 w-4" /> Count Products
   323→                </Button>
   324→              )}
   325→              {!isPaused && (
   326→                <Button variant="outline" size="sm" className="rounded-full text-amber-600 hover:bg-amber-50 hover:text-amber-700 dark:hover:bg-amber-500/10" onClick={() => openPause(open)}>
   327→                  <PauseCircle className="h-4 w-4" /> Pause
   328→                </Button>
   329→              )}
   330→              <Button variant="outline" size="sm" className="rounded-full" onClick={() => openComplete(open)}>
   331→                <ListChecks className="h-4 w-4" /> Review &amp; Complete
   332→              </Button>
   333→              <Button variant="outline" size="sm" className="rounded-full text-red-600 hover:bg-red-50 hover:text-red-700 dark:hover:bg-red-500/10" onClick={() => openCancel(open)}>
   334→                <XCircle className="h-4 w-4" /> Cancel
   335→              </Button>
   336→            </div>
   337→          </div>
   338→
   339→          <div className="grid gap-4 p-5 sm:grid-cols-2 lg:grid-cols-4">
   340→            <div className="rounded-xl border bg-card p-4">
   341→              <p className="text-xs text-muted-foreground">Lines Counted</p>
   342→              <p className="text-xl font-bold tabular-nums">{open.countedItems}<span className="text-sm font-normal text-muted-foreground"> / {open.totalItems}</span></p>
   343→              <Progress value={open.totalItems > 0 ? (open.countedItems / open.totalItems) * 100 : 0} className="mt-2 h-1.5" aria-label="Count progress" />
   344→            </div>
   345→            <div className="rounded-xl border bg-card p-4">
   346→              <p className="text-xs text-muted-foreground">Units — System vs Counted</p>
   347→              <p className="text-xl font-bold tabular-nums">{open.systemQty} <span className="text-sm font-normal text-muted-foreground">→ {open.countedQty}</span></p>
   348→              <p className="mt-0.5 text-[11px] text-muted-foreground">{(open.resumeCount ?? 0) > 0 ? 're-baselined to current stock on resume' : 'snapshot at session start → physical'}</p>
   349→            </div>
   350→            <div className="rounded-xl border bg-card p-4">
   351→              <p className="text-xs text-muted-foreground">Net Quantity Variance</p>
   352→              <p className={`text-xl font-bold tabular-nums ${open.varianceQty > 0 ? 'text-emerald-600 dark:text-emerald-400' : open.varianceQty < 0 ? 'text-red-600 dark:text-red-400' : ''}`}>
   353→                {signedQty(open.varianceQty)} <span className="text-xs font-normal text-muted-foreground">units</span>
   354→              </p>
   355→              <div className="mt-1.5 flex gap-1.5">
   356→                <span className="rounded-full bg-emerald-50 px-2 py-0.5 text-[10px] font-semibold text-emerald-700 dark:bg-emerald-500/10 dark:text-emerald-400">+{open.positiveLines} lines</span>
   357→                <span className="rounded-full bg-red-50 px-2 py-0.5 text-[10px] font-semibold text-red-700 dark:bg-red-500/10 dark:text-red-400">−{open.negativeLines} lines</span>
   358→              </div>
   359→            </div>
   360→            <div className="rounded-xl border bg-card p-4">
   361→              <p className="text-xs text-muted-foreground">Value Impact (at cost)</p>
   362→              <p className={`text-xl font-bold tabular-nums ${open.varianceValue >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-red-600 dark:text-red-400'}`}>
   363→                {open.varianceValue >= 0 ? '+' : '−'}{fmtMoney(Math.abs(open.varianceValue))}
   364→              </p>
   365→              <p className="mt-0.5 text-[11px] text-muted-foreground">{open.matchedLines} lines matched exactly</p>
   366→            </div>
   367→          </div>
   368→        </Card>
   369→      ) : (
   370→        <Card className="flex flex-col items-center gap-4 border-dashed p-10 text-center">
   371→          <span className="flex h-16 w-16 items-center justify-center rounded-2xl bg-teal-600/10 text-teal-600 dark:text-teal-400">
   372→            <ClipboardCheck className="h-8 w-8" />
   373→          </span>
   374→          <div>
   375→            <h2 className="text-lg font-bold">No count session running</h2>
   376→            <p className="mx-auto mt-1 max-w-xl text-sm text-muted-foreground">
   377→              Start a stock take to compare physical shelf quantities against system stock. The session snapshots today&apos;s
   378→              system stock, you enter what you actually count, and every line shows its variance — positive (surplus) or
   379→              negative (shortage) — valued at cost. On completion variances can be applied to system stock with a full audit trail.
   380→            </p>
   381→          </div>
   382→          <Button className="rounded-full bg-teal-600 hover:bg-teal-700" size="lg" onClick={() => setStartOpen(true)}>
   383→            <PlayCircle className="h-5 w-5" /> Start Stock Take
   384→          </Button>
   385→        </Card>
   386→      )}
   387→
   388→      {/* ============ History / Report tabs ============ */}
   389→      <Tabs value={tab} onValueChange={(v) => setTab(v as 'history' | 'report')}>
   390→        <TabsList className="rounded-full">
   391→          <TabsTrigger value="history" className="gap-1.5 rounded-full"><History className="h-3.5 w-3.5" /> Count History</TabsTrigger>
   392→          <TabsTrigger value="report" className="gap-1.5 rounded-full"><BarChart3 className="h-3.5 w-3.5" /> Variance Report</TabsTrigger>
   393→        </TabsList>
   394→      </Tabs>
   395→
   396→      {tab === 'history' && (
   397→        <div className="space-y-3">
